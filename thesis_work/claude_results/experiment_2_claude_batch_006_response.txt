import sys

def solve(n, a, s):
    basis = {}  # leadbit -> value
    for i in range(n-1, -1, -1):
        v = a[i]
        for bit in range(63, -1, -1):
            if (v >> bit) & 1:
                if bit in basis:
                    v ^= basis[bit]
                else:
                    break
        if v != 0:
            lb = v.bit_length() - 1
            basis[lb] = v
            if s[i] == '1':
                return 1
    return 0

            int mid = (lo + hi) >>> 1;

            else hi = mid;
