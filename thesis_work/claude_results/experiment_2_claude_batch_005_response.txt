    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        StringBuilder out = new StringBuilder();
        for (int i = 0; i < n; i++) {
            String s = br.readLine().trim();
            out.append(parseRecord(s)).append('\n');
        }
        System.out.print(out);
    }

    static String parseRecord(String s) {
        Map<Character, Character> next = new HashMap<>();
        Map<Character, Character> prev = new HashMap<>();
        Set<Character> known = new HashSet<>();
        char first = s.charAt(0);
        known.add(first);
        char curPrev = first;
        int i = 1;
        while (i < s.length()) {
            String dir;
            if (i + 1 < s.length() && s.charAt(i) == '-' && s.charAt(i+1) == '>') {
                dir = "back";
                i += 2;
            } else {
                dir = "fwd";
            }
            char cur = s.charAt(i);
            i++;
            known.add(cur);
            if (dir.equals("back")) {
                next.put(curPrev, cur);
                prev.put(cur, curPrev);
            } else {
                prev.put(curPrev, cur);
                next.put(cur, curPrev);
            }
            curPrev = cur;
        }
        char head = 0;
        for (char c : known) {
            if (!prev.containsKey(c)) { head = c; break; }
        }
        StringBuilder sb = new StringBuilder();
        Character cur = head;
        while (cur != null) {
            sb.append((char) cur);
            cur = next.get(cur);
        }
        return sb.toString();
    }
}

string parseRecord(const string &s) {
    map<char, char> nxt, prv;
    set<char> known;
    char first = s[0];
    known.insert(first);
    char curPrev = first;
    int i = 1;
    int len = s.size();
    while (i < len) {
        bool back = false;
        if (i + 1 < len && s[i] == '-' && s[i+1] == '>') {
            back = true;
            i += 2;
        }
        char cur = s[i];
        i++;
        known.insert(cur);
        if (back) {
            nxt[curPrev] = cur;
            prv[cur] = curPrev;
        } else {
            prv[curPrev] = cur;
            nxt[cur] = curPrev;
        }
        curPrev = cur;
    }
    char head = 0;
    for (char c : known) {
        if (prv.find(c) == prv.end()) { head = c; break; }
    }
    string result;
    char cur = head;
    while (true) {
        result += cur;
        auto it = nxt.find(cur);
        if (it == nxt.end()) break;
        cur = it->second;
    }
    return result;
}

            right = 2*node+1

import java.io.*;
import java.util.*;

public class Sol {
    static long[] sumTree;
    static Long[] lazy;
    static int size;

    static void pushDown(int node, int nodeLen) {
        if (lazy[node] != null) {
            long val = lazy[node];
            int left = 2*node, right = 2*node+1;
            int half = nodeLen / 2;
            sumTree[left] = val * half;
            sumTree[right] = val * half;
            lazy[left] = val;
            lazy[right] = val;
            lazy[node] = null;
        }
    }

    static void update(int node, int nodeL, int nodeR, int s, int t, long x) {
        if (t < nodeL || nodeR < s) return;
        if (s <= nodeL && nodeR <= t) {
            sumTree[node] = x * (nodeR - nodeL + 1);
            lazy[node] = x;
            return;
        }
        pushDown(node, nodeR - nodeL + 1);
        int mid = (nodeL + nodeR) / 2;
        update(2*node, nodeL, mid, s, t, x);
        update(2*node+1, mid+1, nodeR, s, t, x);
        sumTree[node] = sumTree[2*node] + sumTree[2*node+1];
    }

    static long query(int node, int nodeL, int nodeR, int s, int t) {
        if (t < nodeL || nodeR < s) return 0;
        if (s <= nodeL && nodeR <= t) return sumTree[node];
        pushDown(node, nodeR - nodeL + 1);
        int mid = (nodeL + nodeR) / 2;
        return query(2*node, nodeL, mid, s, t) + query(2*node+1, mid+1, nodeR, s, t);
    }

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StreamTokenizer st = new StreamTokenizer(br);
        st.nextToken(); int n = (int) st.nval;

        while (size < n) size *= 2;
        sumTree = new long[2 * size];
        lazy = new Long[2 * size];

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < q; i++) {
            st.nextToken(); int typ = (int) st.nval;
            if (typ == 0) {
                st.nextToken(); int s = (int) st.nval;
                st.nextToken(); int t = (int) st.nval;
                st.nextToken(); int x = (int) st.nval;

void update(int node, int nodeL, int nodeR, int s, int t, long long x) {
    if (t < nodeL || nodeR < s) return;
    if (s <= nodeL && nodeR <= t) {
        sumTree[node] = x * (long long)(nodeR - nodeL + 1);
        lazy[node] = x;
        hasLazy[node] = true;
        return;
    }
    pushDown(node, nodeR - nodeL + 1);
    int mid = (nodeL + nodeR) / 2;
    update(2*node, nodeL, mid, s, t, x);
    update(2*node+1, mid+1, nodeR, s, t, x);
    sumTree[node] = sumTree[2*node] + sumTree[2*node+1];
}

long long query(int node, int nodeL, int nodeR, int s, int t) {
    if (t < nodeL || nodeR < s) return 0;
    if (s <= nodeL && nodeR <= t) return sumTree[node];
    pushDown(node, nodeR - nodeL + 1);
    int mid = (nodeL + nodeR) / 2;
    return query(2*node, nodeL, mid, s, t) + query(2*node+1, mid+1, nodeR, s, t);
}

    cin >> n >> q;
    sizeN = 1;
    while (sizeN < n) sizeN *= 2;
    sumTree.assign(2*sizeN, 0);
    lazy.assign(2*sizeN, 0);
    hasLazy.assign(2*sizeN, false);

    string out;

        int typ;

        if (typ == 0) {
            int s, t;
            long long x;
            cin >> s >> t >> x;
            update(1, 0, sizeN-1, s, t, x);
        } else {
            int s, t;
            cin >> s >> t;
            out += to_string(query(1, 0, sizeN-1, s, t)) + "\n";
        }
    }
    cout << out;
    return 0;
}
