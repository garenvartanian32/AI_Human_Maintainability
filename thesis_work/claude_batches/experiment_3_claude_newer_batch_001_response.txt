                best = Math.min(best, above - target);

class Solution:
    def maximumScore(self, nums, k):
        MOD = 10 ** 9 + 7
        n = len(nums)
        limit = max(nums) + 1

        # omega[v] = number of distinct primes dividing v
        omega = [0] * limit
        for p in range(2, limit):
            if omega[p] == 0:
                for multiple in range(p, limit, p):
                    omega[multiple] += 1

        score = [omega[v] for v in nums]

        # nearest strictly-greater score on the left, >= on the right
        left = [-1] * n
        right = [n] * n
        stack = []
        for i in range(n):
            while stack and score[stack[-1]] < score[i]:
                stack.pop()
            left[i] = stack[-1] if stack else -1
            stack.append(i)
        stack.clear()
        for i in reversed(range(n)):
            while stack and score[stack[-1]] <= score[i]:
                stack.pop()
            right[i] = stack[-1] if stack else n
            stack.append(i)

        result = 1
        budget = k
        for i in sorted(range(n), key=lambda idx: nums[idx], reverse=True):
            if budget == 0:
                break
            reach = (i - left[i]) * (right[i] - i)
            use = min(reach, budget)
            result = result * pow(nums[i], use, MOD) % MOD
            budget -= use
        return result

import java.util.*;

class Solution {
    private static final long MOD = 1_000_000_007L;

    public int maximumScore(List<Integer> nums, int k) {
        int n = nums.size();
        int limit = 0;
        for (int v : nums) {
            limit = Math.max(limit, v);
        }
        limit++;

        int[] omega = new int[limit];
        for (int p = 2; p < limit; p++) {
            if (omega[p] == 0) {
                for (int multiple = p; multiple < limit; multiple += p) {
                    omega[multiple]++;
                }
            }
        }

        int[] score = new int[n];
        for (int i = 0; i < n; i++) {
            score[i] = omega[nums.get(i)];
        }

        int[] left = new int[n];
        int[] right = new int[n];
        Deque<Integer> stack = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            while (!stack.isEmpty() && score[stack.peek()] < score[i]) {
                stack.pop();
            }
            left[i] = stack.isEmpty() ? -1 : stack.peek();
            stack.push(i);
        }
        stack.clear();
        for (int i = n - 1; i >= 0; i--) {
            while (!stack.isEmpty() && score[stack.peek()] <= score[i]) {
                stack.pop();
            }
            right[i] = stack.isEmpty() ? n : stack.peek();
            stack.push(i);
        }

        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) {
            order[i] = i;
        }
        Arrays.sort(order, (a, b) -> Integer.compare(nums.get(b), nums.get(a)));

        long result = 1;
        long budget = k;
        for (int i : order) {
            if (budget == 0) {
                break;
            }
            long reach = (long) (i - left[i]) * (right[i] - i);
            long use = Math.min(reach, budget);
            result = result * modPow(nums.get(i), use) % MOD;
            budget -= use;
        }
        return (int) result;
    }

    private long modPow(long base, long exp) {
        long acc = 1;
        base %= MOD;
        while (exp > 0) {
            if ((exp & 1L) == 1L) {
                acc = acc * base % MOD;
            }
            base = base * base % MOD;
            exp >>= 1;
        }
        return acc;
    }
}

class Solution {
    public long findMaximumElegance(int[][] items, int k) {
        Arrays.sort(items, (a, b) -> Integer.compare(b[0], a[0]));

        long profit = 0;
        Set<Integer> seen = new HashSet<>();
        Deque<Integer> spare = new ArrayDeque<>();
        for (int i = 0; i < k; i++) {
            profit += items[i][0];
            if (!seen.add(items[i][1])) {
                spare.push(items[i][0]);
            }
        }

        long distinct = seen.size();
        long best = profit + distinct * distinct;

            if (spare.isEmpty()) {
                break;
            }
            if (seen.contains(items[i][1])) {
                continue;
            }
            profit += items[i][0] - spare.pop();
            seen.add(items[i][1]);
            distinct = seen.size();
            best = Math.max(best, profit + distinct * distinct);
        }
        return best;
    }

import sys


class Fenwick:
    def __init__(self, size):
        self.size = size
        self.cnt = [0] * (size + 1)
        self.tot = [0] * (size + 1)

    def add(self, i, x):
        while i <= self.size:
            self.cnt[i] += 1
            self.tot[i] += x
            i += i & -i

    def query(self, i):
        c = t = 0
        while i > 0:
            c += self.cnt[i]
            t += self.tot[i]
            i -= i & -i
        return c, t


def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    xs = list(map(int, data[1:1 + n]))
    vs = list(map(int, data[1 + n:1 + 2 * n]))

    ranks = {v: i + 1 for i, v in enumerate(sorted(set(vs)))}
    tree = Fenwick(len(ranks))

    total = 0
    for x, v in sorted(zip(xs, vs)):
        c, t = tree.query(ranks[v])
        total += c * x - t
        tree.add(ranks[v], x)

    print(total)
