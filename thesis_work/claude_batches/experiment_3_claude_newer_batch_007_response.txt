                cost[g][y] = sum;

def main():
    data = sys.stdin.buffer.read().split()
    x1, y1, x2, y2 = (int(data[i]) for i in range(4))
    n = int(data[4])

    fountains = []
    pos = 5
    for _ in range(n):
        fountains.append((int(data[pos]), int(data[pos + 1])))
        pos += 2

    radius = 10.0
    walk = (abs(x1 - x2) + abs(y1 - y2)) * 100.0

    if x1 == x2 or y1 == y2:
        blocked = False
        if y1 == y2:
            lo, hi = sorted((x1, x2))
            blocked = any(fy == y1 and lo < fx < hi for fx, fy in fountains)
        else:
            lo, hi = sorted((y1, y2))
            blocked = any(fx == x1 and lo < fy < hi for fx, fy in fountains)
        detour = radius * (pi - 2) if blocked else 0.0
        print('%.15f' % (walk + detour))
        return

    xlo, xhi = sorted((x1, x2))
    ylo, yhi = sorted((y1, y2))
    xdir = 1 if x2 > x1 else -1
    ydir = 1 if y2 > y1 else -1

    inside = sorted(
        (fx * xdir, fy * ydir)
        for fx, fy in fountains
        if xlo < fx < xhi and ylo < fy < yhi
    )

    tails = []
    for _, ty in inside:
        slot = bisect_left(tails, ty)
        if slot == len(tails):
            tails.append(ty)
        else:
            tails[slot] = ty

    bonus = radius * (2 - pi / 2)
    print('%.15f' % (walk - len(tails) * bonus))

    for (const auto &row : grid) {

def intersection(interval1, interval2):
    """You are given two intervals,
    where each interval is a pair of integers. For example, interval = (start, end) = (1, 2).
    The given intervals are closed which means that the interval (start, end)
    includes both start and end.
    For each given interval, it is assumed that its start is less or equal its end.
    Your task is to determine whether the length of intersection of these two 
    intervals is a prime number.
    Example, the intersection of the intervals (1, 3), (2, 4) is (2, 3)
    which its length is 1, which not a prime number.
    If the length of the intersection is a prime number, return "YES",
    otherwise, return "NO".
    If the two intervals don't intersect, return "NO".


    [input/output] samples:
    intersection((1, 2), (2, 3)) ==> "NO"
    intersection((-1, 1), (0, 4)) ==> "NO"
    intersection((-3, -1), (-5, 5)) ==> "YES"
    """
    span = min(interval1[1], interval2[1]) - max(interval1[0], interval2[0])
    if span < 2:
        return "NO"
    divisor = 2
    while divisor * divisor <= span:
        if span % divisor == 0:
            return "NO"
        divisor += 1
    return "YES"
