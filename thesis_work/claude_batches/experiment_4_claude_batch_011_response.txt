def solve(n, a, s):
    basis = {}  # leadbit -> value
    for i in range(n-1, -1, -1):
        v = a[i]
        for bit in range(63, -1, -1):
            if (v >> bit) & 1:
                if bit in basis:
                    v ^= basis[bit]
                else:
                    break
        if v != 0:
            lb = v.bit_length() - 1
            basis[lb] = v
            if s[i] == '1':
                return 1
    return 0

"""Decide whether the first player can win the XOR-basis game."""

import sys

BIT_WIDTH = 64
FIRST_PLAYER_WINS = 1
FIRST_PLAYER_LOSES = 0
FIRST_PLAYER_MARK = "1"


def first_player_wins(values: list[int], owners: str) -> int:
    """True when the first player owns a value that is linearly independent.

    Values are folded into a linear basis over GF(2) from the back of the list
    forwards. The first value that survives reduction is the deciding one: if
    the first player owns it, they can force the win.
    """
    basis = [0] * BIT_WIDTH

    for index in reversed(range(len(values))):
        residue = values[index]

        for bit in reversed(range(BIT_WIDTH)):
            if not (residue >> bit) & 1:
                continue
            if basis[bit] == 0:
                basis[bit] = residue
                return (
                    FIRST_PLAYER_WINS
                    if owners[index] == FIRST_PLAYER_MARK
                    else FIRST_PLAYER_LOSES
                )
            residue ^= basis[bit]

    return FIRST_PLAYER_LOSES

 * forwards. The first value that survives reduction is the deciding one.

    unsigned long long basis[kBitWidth] = {0};

}

"""List every length-n string in normal form, in lexicographic order."""

import sys

FIRST_LETTER = "a"


def normal_form_strings(length: int) -> list[str]:
    """All strings where each letter is at most one past the largest so far.

    That rule keeps exactly one representative of every way of partitioning the
    positions, and generating letters in alphabetical order emits the strings
    already sorted.
    """
    results: list[str] = []
    buffer = [FIRST_LETTER] * length

    def extend(position: int, letters_used: int) -> None:
        if position == length:
            results.append("".join(buffer))
            return

        for letter_index in range(letters_used + 1):
            buffer[position] = chr(ord(FIRST_LETTER) + letter_index)
            next_used = letters_used + 1 if letter_index == letters_used else letters_used
            extend(position + 1, next_used)

    extend(0, 0)
    return results


def main() -> None:
    length = int(sys.stdin.read().strip())
    sys.stdout.write("\n".join(normal_form_strings(length)) + "\n")

import java.io.IOException;
import java.io.InputStreamReader;

/**
 * Lists every length-n string in normal form, in lexicographic order.
 *
 * <p>Each letter may be at most one past the largest used so far. That rule
 * keeps exactly one representative of every way of partitioning the positions,
 * and generating letters in alphabetical order emits them already sorted.
 */
public class Sol {

    private static final char FIRST_LETTER = 'a';

    private static int length;
    private static char[] buffer;
    private static StringBuilder results;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        length = Integer.parseInt(reader.readLine().trim());

        buffer = new char[length];
        results = new StringBuilder();

        extend(0, 0);
        System.out.print(results);
    }

    private static void extend(int position, int lettersUsed) {
        if (position == length) {
            results.append(buffer, 0, length).append('\n');
            return;
        }

        for (int letterIndex = 0; letterIndex <= lettersUsed; letterIndex++) {
            buffer[position] = (char) (FIRST_LETTER + letterIndex);
            int nextUsed = (letterIndex == lettersUsed) ? lettersUsed + 1 : lettersUsed;
            extend(position + 1, nextUsed);
        }
    }
}
