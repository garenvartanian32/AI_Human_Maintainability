        long span = Math.min(interval1.getValue1(), interval2.getValue1())
                  - Math.max(interval1.getValue0(), interval2.getValue0());
        if (span < 2) return "NO";
        for (long divisor = 2; divisor * divisor <= span; divisor++) {
            if (span % divisor == 0) return "NO";
        }
        return "YES";
    }

    after making the ordered lists of the values on the cells that A and B go

    than lst_B, in other words, there exist an integer index i (1 <= i <= k)
    such that lst_A[i] < lst_B[i] and for any j (1 <= j < i) we have

    It is guaranteed that the answer is unique.

        Input: grid = [ [1,2,3], [4,5,6], [7,8,9]], k = 3
        Output: [1, 2, 1]

        Output: [1]

        for c in range(size):
            if grid[r][c] == 1:

        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1))
        if 0 <= row + dr < size and 0 <= col + dc < size
    )

    return [1 if step % 2 == 0 else partner for step in range(k)]

        elif i % 2 == 0:
            sequence.append(even_term(i))

    operator['+', '*', '-']

    public static long doAlgebra(ArrayList<String> op, ArrayList<Long> operand) {

        LinkedList<String> ops = new LinkedList<>(op);

        collapse(values, ops, new HashSet<>(Arrays.asList("**")));
        collapse(values, ops, new HashSet<>(Arrays.asList("*", "//")));
        collapse(values, ops, new HashSet<>(Arrays.asList("+", "-")));

    private static void collapse(LinkedList<Long> values, LinkedList<String> ops, Set<String> wanted) {
        int i = 0;
        while (i < ops.size()) {
            String o = ops.get(i);
            if (!wanted.contains(o)) {
                i++;
                continue;
            }
            long a = values.get(i), b = values.get(i + 1);
            long result;
            switch (o) {
                case "**": result = power(a, b); break;
                case "*": result = a * b; break;
                case "//": result = Math.floorDiv(a, b); break;
                case "+": result = a + b; break;
                default: result = a - b; break;
            }
            values.set(i, result);
            values.remove(i + 1);
            ops.remove(i);
        }
    }

    private static long power(long base, long exp) {
        long acc = 1;
        for (long e = 0; e < exp; e++) acc *= base;
        return acc;
    }

static long apply_op(const std::string &o, long a, long b) {
    if (o == "**") {
        long acc = 1;
        for (long e = 0; e < b; e++) acc *= a;
        return acc;
    }
    if (o == "*") return a * b;
    if (o == "//") {
        long q = a / b;
        if ((a % b != 0) && ((a < 0) != (b < 0))) q--;
        return q;
    }
    if (o == "+") return a + b;
    return a - b;
}

            i++;
            continue;
        }
        values[i] = apply_op(ops[i], values[i], values[i + 1]);
        values.erase(values.begin() + i + 1);
        ops.erase(ops.begin() + i);
    }
}

    public static String solve(String s) {

            if (Character.isLetter(s.charAt(i))) { hasLetter = true; break; }
        }
        if (!hasLetter) {

        char[] out = s.toCharArray();

            else if (Character.isLowerCase(out[i])) out[i] = Character.toUpperCase(out[i]);
        }
        return new String(out);
    }

                                 [](unsigned char ch) { return std::isalpha(ch); });

        std::reverse(s.begin(), s.end());
        return s;

        if (std::isupper(u)) ch = (char) std::tolower(u);

    }
