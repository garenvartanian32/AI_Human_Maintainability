        boolean sawOddDigit = false;

        List<Long> values = new ArrayList<>(operand);

                long res = 1;
                for (long e = 0; e < exp; e++) res *= base;
                values.set(i, res);
                values.remove(i + 1);
                ops.remove(i);
            } else {
                i++;
            }

        // Pass 2: * and //
        for (int i = 0; i < ops.size(); ) {
            String o = ops.get(i);
            if (o.equals("*") || o.equals("//")) {
                long a = values.get(i), b = values.get(i + 1);

                values.set(i, res);
                values.remove(i + 1);
                ops.remove(i);

                i++;
            }

        long result = values.get(0);
        for (int i = 0; i < ops.size(); i++) {
            String o = ops.get(i);
            long b = values.get(i + 1);
            if (o.equals("+")) result += b;
            else if (o.equals("-")) result -= b;
        }
        return result;

class Problem {
    // Given two array lists operator, and operand. The first array list has basic algebra operations, and 
    // the second array list is an array array list of integers. Use the two given array lists to build the algebric 
    // expression and return the evaluation of this expression.
    // The basic algebra operations:
    // Addition ( + ) 
    // Subtraction ( - ) 
    // Multiplication ( * ) 
    // Floor division ( // ) 
    // Exponentiation ( ** ) 
    // Example:
    // operator['+', '*', '-']
    // array array list = [2, 3, 4, 5]
    // result = 2 + 3 * 4 - 5
    // => result = 9
    // Note:
    // The length of operator array list is equal to the length of operand array list minus one.
    // Operand is an array array list of of non-negative integers.
    // Operator array list has at least one operator, and operand array list has at least two operands.

    // Operators are collapsed in precedence order: exponentiation binds
    // tightest, then multiplication and floor division, then the additive pair.
    private static final List<Set<String>> PRECEDENCE_LEVELS = Arrays.asList(
            new HashSet<>(Arrays.asList("**")),
            new HashSet<>(Arrays.asList("*", "//")),
            new HashSet<>(Arrays.asList("+", "-")));

    public static long doAlgebra(ArrayList<String> op, ArrayList<Long> operand) {
        LinkedList<Long> values = new LinkedList<>(operand);
        LinkedList<String> symbols = new LinkedList<>(op);

        for (Set<String> level : PRECEDENCE_LEVELS) {
            collapse(values, symbols, level);
        }

        return values.getFirst();
    }

    /** Applies every operator belonging to {@code level}, left to right. */
    private static void collapse(
            LinkedList<Long> values, LinkedList<String> symbols, Set<String> level) {
        int position = 0;
        while (position < symbols.size()) {
            String symbol = symbols.get(position);
            if (!level.contains(symbol)) {
                position++;
                continue;
            }

            long result = apply(symbol, values.get(position), values.get(position + 1));
            values.set(position, result);
            values.remove(position + 1);
            symbols.remove(position);
        }
    }

    private static long apply(String symbol, long left, long right) {
        switch (symbol) {
            case "**":
                return power(left, right);
            case "*":
                return left * right;
            case "//":
                return Math.floorDiv(left, right);
            case "+":
                return left + right;
            default:
                return left - right;
        }
    }

    private static long power(long base, long exponent) {
        long result = 1;
        for (long step = 0; step < exponent; step++) {
            result *= base;
        }
        return result;
    }

/// Applies one operator to its two operands.
static long applyOperator(const std::string& symbol, long left, long right) {
    if (symbol == "**") {
        long result = 1;
        for (long step = 0; step < right; ++step) {
            result *= left;
        }
        return result;
    }
    if (symbol == "*") {
        return left * right;
    }
    if (symbol == "//") {
        long quotient = left / right;
        if ((left % right != 0) && ((left < 0) != (right < 0))) {
            quotient--;  // C++ truncates towards zero; Python floors.
        }
        return quotient;
    }
    if (symbol == "+") {
        return left + right;
    }
    return left - right;
}

/// Applies every operator belonging to `level`, left to right.
static void collapse(std::vector<long>& values, std::vector<std::string>& symbols,
                     const std::set<std::string>& level) {
    std::size_t position = 0;
    while (position < symbols.size()) {
        if (level.count(symbols[position]) == 0) {
            ++position;
            continue;
        }

        values[position] = applyOperator(symbols[position], values[position], values[position + 1]);
        values.erase(values.begin() + position + 1);
        symbols.erase(symbols.begin() + position);
    }
}

    const std::vector<std::set<std::string>> precedenceLevels = {
        {"**"}, {"*", "//"}, {"+", "-"}};

    for (const auto& level : precedenceLevels) {
        collapse(values, symbols, level);
    }

    return values[0];

        }

    private static boolean containsLetter(String text) {
        for (int i = 0; i < text.length(); i++) {
            if (Character.isLetter(text.charAt(i))) {
                return true;
            }
        }
        return false;
    }

    /** Flips the case of every letter, leaving other characters untouched. */
    private static String swapCase(String text) {
        char[] flipped = text.toCharArray();
        for (int i = 0; i < flipped.length; i++) {
            if (Character.isUpperCase(flipped[i])) {
                flipped[i] = Character.toLowerCase(flipped[i]);
            } else if (Character.isLowerCase(flipped[i])) {
                flipped[i] = Character.toUpperCase(flipped[i]);
            }
        }
        return new String(flipped);
    }
