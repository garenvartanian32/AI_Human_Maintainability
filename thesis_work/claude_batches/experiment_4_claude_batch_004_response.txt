"""A toy memory manager supporting alloc, erase and defragment."""

import sys

FREE = 0
NULL_RESULT = "NULL"
ILLEGAL_ERASE = "ILLEGAL_ERASE_ARGUMENT"


class MemoryManager:
    """Fixed-size memory modelled as a list of owning block identifiers."""

    def __init__(self, size: int) -> None:
        self._cells = [FREE] * size
        self._live_blocks: set[int] = set()
        self._next_identifier = 1

    def allocate(self, cells_needed: int) -> str:
        """Reserve the first run of free cells that is long enough."""
        start = self._find_free_run(cells_needed)
        if start is None:
            return NULL_RESULT

        identifier = self._next_identifier
        self._next_identifier += 1

        for cell in range(start, start + cells_needed):
            self._cells[cell] = identifier
        self._live_blocks.add(identifier)

        return str(identifier)

    def erase(self, identifier: int) -> str | None:
        """Free a live block; returns an error string for unknown blocks."""
        if identifier not in self._live_blocks:
            return ILLEGAL_ERASE

        self._live_blocks.discard(identifier)
        for cell, owner in enumerate(self._cells):
            if owner == identifier:
                self._cells[cell] = FREE
        return None

    def defragment(self) -> None:
        """Slide every allocated block towards the start, preserving order."""
        occupied = [owner for owner in self._cells if owner != FREE]
        self._cells = occupied + [FREE] * (len(self._cells) - len(occupied))

    def _find_free_run(self, cells_needed: int) -> int | None:
        run_length = 0
        for cell, owner in enumerate(self._cells):
            run_length = run_length + 1 if owner == FREE else 0
            if run_length == cells_needed:
                return cell - cells_needed + 1
        return None


def main() -> None:
    lines = sys.stdin.read().split("\n")
    operation_count, memory_size = (int(value) for value in lines[0].split())

    manager = MemoryManager(memory_size)
    output: list[str] = []

    for index in range(1, operation_count + 1):
        parts = lines[index].split()
        if not parts:
            continue

        command = parts[0]
        if command == "alloc":
            output.append(manager.allocate(int(parts[1])))
        elif command == "erase":
            error = manager.erase(int(parts[1]))
            if error is not None:
                output.append(error)
        elif command == "defragment":
            manager.defragment()

    sys.stdout.write("\n".join(output) + ("\n" if output else ""))

/**
 * A toy memory manager supporting alloc, erase and defragment.
 */
public class Sol {

    private static final int FREE = 0;
    private static final String NULL_RESULT = "NULL";
    private static final String ILLEGAL_ERASE = "ILLEGAL_ERASE_ARGUMENT";
    private static final int NO_RUN_FOUND = -1;

    /** Fixed-size memory modelled as an array of owning block identifiers. */
    private static final class MemoryManager {
        private final int[] cells;
        private final Set<Integer> liveBlocks = new HashSet<>();
        private int nextIdentifier = 1;

        MemoryManager(int size) {
            this.cells = new int[size];
        }

        /** Reserves the first run of free cells that is long enough. */
        String allocate(int cellsNeeded) {
            int start = findFreeRun(cellsNeeded);
            if (start == NO_RUN_FOUND) {
                return NULL_RESULT;
            }

            int identifier = nextIdentifier++;
            Arrays.fill(cells, start, start + cellsNeeded, identifier);
            liveBlocks.add(identifier);
            return Integer.toString(identifier);
        }

        /** Frees a live block; returns an error string for unknown blocks. */
        String erase(int identifier) {
            if (!liveBlocks.remove(identifier)) {
                return ILLEGAL_ERASE;
            }
            for (int cell = 0; cell < cells.length; cell++) {
                if (cells[cell] == identifier) {
                    cells[cell] = FREE;
                }
            }
            return null;
        }

        /** Slides every allocated block towards the start, preserving order. */
        void defragment() {
            int writePosition = 0;
            for (int cell = 0; cell < cells.length; cell++) {
                if (cells[cell] != FREE) {
                    cells[writePosition++] = cells[cell];
                }
            }
            Arrays.fill(cells, writePosition, cells.length, FREE);
        }

        private int findFreeRun(int cellsNeeded) {
            int runLength = 0;
            for (int cell = 0; cell < cells.length; cell++) {
                runLength = (cells[cell] == FREE) ? runLength + 1 : 0;
                if (runLength == cellsNeeded) {
                    return cell - cellsNeeded + 1;
                }
            }
            return NO_RUN_FOUND;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer header = new StringTokenizer(reader.readLine());
        int operationCount = Integer.parseInt(header.nextToken());
        int memorySize = Integer.parseInt(header.nextToken());

        MemoryManager manager = new MemoryManager(memorySize);
        StringBuilder output = new StringBuilder();

        for (int operation = 0; operation < operationCount; operation++) {
            StringTokenizer line = new StringTokenizer(reader.readLine());
            String command = line.nextToken();

            switch (command) {
                case "alloc":
                    output.append(manager.allocate(Integer.parseInt(line.nextToken())))
                          .append('\n');
                    break;
                case "erase":
                    String error = manager.erase(Integer.parseInt(line.nextToken()));
                    if (error != null) {
                        output.append(error).append('\n');
                    }
                    break;
                case "defragment":
                    manager.defragment();
                    break;
                default:
                    break;
            }
        }

        System.out.print(output);
    }

constexpr int kFree = 0;
constexpr int kNoRunFound = -1;
const std::string kNullResult = "NULL";
const std::string kIllegalErase = "ILLEGAL_ERASE_ARGUMENT";

/// Fixed-size memory modelled as a vector of owning block identifiers.
class MemoryManager {
public:
    explicit MemoryManager(int size) : cells_(size, kFree) {}

    /// Reserves the first run of free cells that is long enough.
    std::string allocate(int cellsNeeded) {
        const int start = findFreeRun(cellsNeeded);
        if (start == kNoRunFound) {
            return kNullResult;
        }

        const int identifier = nextIdentifier_++;
        std::fill(cells_.begin() + start, cells_.begin() + start + cellsNeeded, identifier);
        liveBlocks_.insert(identifier);
        return std::to_string(identifier);
    }

    /// Frees a live block; reports whether an error message is needed.
    bool erase(int identifier, std::string* error) {
        if (liveBlocks_.erase(identifier) == 0) {
            *error = kIllegalErase;
            return false;
        }
        for (int& owner : cells_) {
            if (owner == identifier) {
                owner = kFree;
            }
        }
        return true;
    }

    /// Slides every allocated block towards the start, preserving order.
    void defragment() {
        std::size_t writePosition = 0;
        for (std::size_t cell = 0; cell < cells_.size(); ++cell) {
            if (cells_[cell] != kFree) {
                cells_[writePosition++] = cells_[cell];
            }
        }
        std::fill(cells_.begin() + writePosition, cells_.end(), kFree);
    }

private:
    int findFreeRun(int cellsNeeded) const {
        int runLength = 0;
        for (std::size_t cell = 0; cell < cells_.size(); ++cell) {
            runLength = (cells_[cell] == kFree) ? runLength + 1 : 0;
            if (runLength == cellsNeeded) {
                return static_cast<int>(cell) - cellsNeeded + 1;
            }
        }
        return kNoRunFound;
    }

    std::vector<int> cells_;
    std::set<int> liveBlocks_;
    int nextIdentifier_ = 1;
};

"""Enumerate every length-n string in normal form, in lexicographic order."""

import sys

FIRST_LETTER = "a"


def normal_form_strings(length: int) -> list[str]:
    """Return all normal-form strings of the given length.

    A string is in normal form when each letter is at most one past the
    largest letter used before it. That keeps exactly one representative of
    every partition of the positions, and building left to right in
    alphabetical order yields them already sorted.
    """
    results: list[str] = []
    buffer = [FIRST_LETTER] * length

    def extend(position: int, highest_used: int) -> None:
        if position == length:
            results.append("".join(buffer))
            return

        for letter_index in range(highest_used + 1):
            buffer[position] = chr(ord(FIRST_LETTER) + letter_index)
            next_highest = highest_used + 1 if letter_index == highest_used else highest_used
            extend(position + 1, next_highest)

    extend(0, 0)
    return results


def main() -> None:
    length = int(sys.stdin.buffer.read().split()[0])
    sys.stdout.write("\n".join(normal_form_strings(length)) + "\n")

import java.io.BufferedReader;

import java.io.InputStreamReader;

/**
 * Enumerates every length-n string in normal form, in lexicographic order.
 *
 * <p>A string is in normal form when each letter is at most one past the largest
 * letter used before it. Building left to right in alphabetical order yields
 * the strings already sorted.
 */
public class Sol {

    private static final char FIRST_LETTER = 'a';

    private static int length;
    private static char[] buffer;
    private static StringBuilder results;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        length = Integer.parseInt(reader.readLine().trim());

        buffer = new char[length];
        results = new StringBuilder();

        extend(0, 0);
        System.out.print(results);
    }

    private static void extend(int position, int highestUsed) {
        if (position == length) {
            results.append(buffer, 0, length).append('\n');
            return;
        }

#include <iostream>
#include <string>

namespace {

constexpr char kFirstLetter = 'a';

int gLength;
std::string gBuffer;
std::string gResults;

/**
 * Extends the buffer at `position`, given the largest letter index used so far.
 *
 * A string is in normal form when each letter is at most one past the largest
 * letter used before it, so building left to right in alphabetical order yields
 * the strings already sorted.
 */
void extend(int position, int highestUsed) {
    if (position == gLength) {
        gResults += gBuffer;
        gResults += '\n';
        return;
    }

    for (int letterIndex = 0; letterIndex <= highestUsed; ++letterIndex) {
        gBuffer[position] = static_cast<char>(kFirstLetter + letterIndex);
        const int nextHighest = (letterIndex == highestUsed) ? highestUsed + 1 : highestUsed;
        extend(position + 1, nextHighest);
    }
}

}  // namespace

int main() {
    std::cin >> gLength;
    gBuffer.assign(gLength, kFirstLetter);

    extend(0, 0);
    std::cout << gResults;
    return 0;
}

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

/**
 * Reports whether the product of two integers is odd or even.
 */
public class Sol {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer line = new StringTokenizer(reader.readLine());

        long first = Long.parseLong(line.nextToken());
        long second = Long.parseLong(line.nextToken());

        // A product is odd only when neither factor contributes a factor of two.
        boolean productIsOdd = isOdd(first) && isOdd(second);
        System.out.println(productIsOdd ? "Odd" : "Even");
    }

    private static boolean isOdd(long value) {
        return value % 2 != 0;
    }
}
