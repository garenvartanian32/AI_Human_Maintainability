import sys
from collections import defaultdict


class DisjointSet:
    """Union-find over positions, with path compression."""

    def __init__(self, size: int) -> None:
        self._parent = list(range(size + 1))

    def find(self, position: int) -> int:
        root = position
        while self._parent[root] != root:
            root = self._parent[root]
        while self._parent[position] != root:
            self._parent[position], position = root, self._parent[position]
        return root

    def union(self, first: int, second: int) -> None:
        first_root, second_root = self.find(first), self.find(second)
        if first_root != second_root:
            self._parent[first_root] = second_root


def main() -> None:
    tokens = sys.stdin.buffer.read().split()

    # can be permuted arbitrarily. Placing them largest-first at the smallest

    positions = DisjointSet(size)

        available = sorted((values[position - 1] for position in member_positions), reverse=True)
        for position, value in zip(member_positions, available):

/**
 * Largest permutation reachable using an allowed set of position swaps.
 *
 * <p>Swaps are freely composable, so within one connected component the values
 * can be permuted arbitrarily. Placing them largest-first at the smallest
 * positions therefore gives the lexicographic maximum.
 */
public class Sol {

    /** Union-find over positions, with path compression. */
    private static final class DisjointSet {
        private final int[] parent;

        DisjointSet(int size) {
            parent = new int[size + 1];
            for (int position = 0; position <= size; position++) {
                parent[position] = position;
            }
        }

        int find(int position) {
            while (parent[position] != position) {
                parent[position] = parent[parent[position]];
                position = parent[position];
            }
            return position;
        }

        void union(int first, int second) {
            int firstRoot = find(first);
            int secondRoot = find(second);
            if (firstRoot != secondRoot) {
                parent[firstRoot] = secondRoot;
            }
        }
    }

    public static void main(String[] args) throws IOException {
        DataInputStream input = new DataInputStream(new BufferedInputStream(System.in, 1 << 16));

        int size = nextInt(input);
        int swapCount = nextInt(input);

        int[] values = new int[size + 1];
        for (int position = 1; position <= size; position++) {
            values[position] = nextInt(input);
        }

        DisjointSet positions = new DisjointSet(size);
        for (int swap = 0; swap < swapCount; swap++) {
            positions.union(nextInt(input), nextInt(input));
        }

        List<List<Integer>> components = groupByComponent(positions, size);

        int[] answer = new int[size + 1];
        for (List<Integer> memberPositions : components) {
            List<Integer> available = new ArrayList<>();
            for (int position : memberPositions) {
                available.add(values[position]);
            }
            available.sort(Collections.reverseOrder());
            for (int i = 0; i < memberPositions.size(); i++) {
                answer[memberPositions.get(i)] = available.get(i);
            }
        }

        StringBuilder output = new StringBuilder();
        for (int position = 1; position <= size; position++) {
            if (position > 1) {
                output.append(' ');
            }
            output.append(answer[position]);
        }
        System.out.println(output);
    }

    private static List<List<Integer>> groupByComponent(DisjointSet positions, int size) {
        int[] componentIndex = new int[size + 1];
        Arrays.fill(componentIndex, -1);

        List<List<Integer>> components = new ArrayList<>();
        for (int position = 1; position <= size; position++) {
            int root = positions.find(position);
            if (componentIndex[root] < 0) {
                componentIndex[root] = components.size();
                components.add(new ArrayList<>());
            }
            components.get(componentIndex[root]).add(position);
        }
        return components;
    }

    private static int nextInt(DataInputStream input) throws IOException {
        int value = 0;
        int character = input.read();
        while (character < '0' || character > '9') {
            character = input.read();
        }
        while (character >= '0' && character <= '9') {
            value = value * 10 + (character - '0');
            character = input.read();
        }

class DisjointSet {
public:
    explicit DisjointSet(int size) : parent_(size + 1) {
        std::iota(parent_.begin(), parent_.end(), 0);
    }

    int find(int position) {
        while (parent_[position] != position) {
            parent_[position] = parent_[parent_[position]];
            position = parent_[position];
        }
        return position;
    }

    void unite(int first, int second) {
        const int firstRoot = find(first);
        const int secondRoot = find(second);
        if (firstRoot != secondRoot) {
            parent_[firstRoot] = secondRoot;
        }
    }

private:
    std::vector<int> parent_;
};

import sys

def parse_record(s):
    prev = s[0]
    prev_node = {'char': prev, 'next': None, 'prev': None}
    nodes = {prev: prev_node}
    i = 1
    while i < len(s):
        if s[i:i+2] == '->':
            direction = 'back'
            i += 2
        else:
            direction = 'fwd'
        cur = s[i]
        i += 1
        if cur not in nodes:
            nodes[cur] = {'char': cur, 'next': None, 'prev': None}
        if direction == 'back':
            nodes[prev]['next'] = cur
            nodes[cur]['prev'] = prev
        else:
            nodes[prev]['prev'] = cur
            nodes[cur]['next'] = prev
        prev = cur
    # find head (prev is None)
    head = None
    for ch, node in nodes.items():
        if node['prev'] is None:
            head = ch
            break
    result = []
    cur = head
    while cur is not None:
        result.append(cur)
        cur = nodes[cur]['next']
    return ''.join(result)

def main():
    data = sys.stdin.read().split('\n')
    n = int(data[0].strip())
    out = []
    for i in range(1, n+1):
        s = data[i].strip()
        out.append(parse_record(s))
    print('\n'.join(out))

main()

"""Reconstruct a train's carriage order from a conductor's walk record."""

import sys

BACKWARD_MARKER = "->"


def rebuild_train(record: str) -> str:
    """Return the carriages from the front, given one walk record.

    The record is a chain of moves; each move links the current carriage to the
    next one either behind it (``->``) or in front of it. Recording both links
    builds a doubly linked list, and the front carriage is the only one with
    nothing ahead of it.
    """
    carriage_behind: dict[str, str] = {}
    carriage_ahead: dict[str, str] = {}
    seen: list[str] = []

    current = record[0]
    seen.append(current)

    position = 1
    while position < len(record):
        moved_backward = record[position : position + len(BACKWARD_MARKER)] == BACKWARD_MARKER
        if moved_backward:
            position += len(BACKWARD_MARKER)

        next_carriage = record[position]
        position += 1
        if next_carriage not in seen:
            seen.append(next_carriage)

        if moved_backward:
            carriage_behind[current] = next_carriage
            carriage_ahead[next_carriage] = current
        else:
            carriage_ahead[current] = next_carriage
            carriage_behind[next_carriage] = current

        current = next_carriage

    front = next(carriage for carriage in seen if carriage not in carriage_ahead)

    order = []
    carriage = front
    while carriage is not None:
        order.append(carriage)
        carriage = carriage_behind.get(carriage)
    return "".join(order)


def main() -> None:
    lines = sys.stdin.read().split("\n")
    record_count = int(lines[0].strip())

    trains = [rebuild_train(lines[index].strip()) for index in range(1, record_count + 1)]
    sys.stdout.write("\n".join(trains) + "\n")


if __name__ == "__main__":
    main()
