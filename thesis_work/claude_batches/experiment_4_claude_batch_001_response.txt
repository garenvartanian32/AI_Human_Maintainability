        members form a valid pair, so the best pair is simply the two

class Solution {

    private static final char REVERSE_KEY = 'i';

    /**
     * Returns the text shown on screen after typing every character of {@code s}.
     *
     * <p>Rather than reversing the buffer on each {@code REVERSE_KEY} (which is
     * quadratic), the buffer is kept in a deque and a flag tracks whether it is
     * currently stored forwards or backwards.
     */
    public String finalString(String s) {
        Deque<Character> characters = new ArrayDeque<>();
        boolean storedForwards = true;

        for (char character : s.toCharArray()) {
            if (character == REVERSE_KEY) {
                storedForwards = !storedForwards;
            } else if (storedForwards) {
                characters.addLast(character);
            } else {
                characters.addFirst(character);
            }
        }

        return joinInOrder(characters, storedForwards);
    }

    /** Concatenates the deque, walking backwards when the buffer is reversed. */
    private String joinInOrder(Deque<Character> characters, boolean storedForwards) {
        StringBuilder builder = new StringBuilder(characters.size());
        Iterator<Character> walker =
                storedForwards ? characters.iterator() : characters.descendingIterator();
        while (walker.hasNext()) {
            builder.append(walker.next());
        }
        return builder.toString();
    }

class Solution:
    def minAbsoluteDifference(self, nums: list[int], x: int) -> int:
        """Return min |nums[i] - nums[j]| over all pairs with |i - j| >= x.

        A sorted window holds every element that is far enough behind the
        current index. For each position we only need the two window entries
        that bracket the current value, found by binary search.
        """
        if x == 0:
            return 0

        eligible_values: list[int] = []
        smallest_difference = float("inf")

        for index in range(x, len(nums)):
            insort(eligible_values, nums[index - x])
            current = nums[index]

            insertion_point = bisect_left(eligible_values, current)
            if insertion_point < len(eligible_values):
                smallest_difference = min(
                    smallest_difference, eligible_values[insertion_point] - current
                )
            if insertion_point > 0:
                smallest_difference = min(
                    smallest_difference, current - eligible_values[insertion_point - 1]
                )

        return smallest_difference

        backwards, so the recursion doubles on the way *out* of the call stack.

        carry_in = self._double_from(node.next)

        return doubled // 10
