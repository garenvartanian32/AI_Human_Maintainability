/**
 * Decodes a message enciphered with a repeating sequence of shift keys.
 *
 * <p>Lower and upper case form one continuous 52-letter ring, so a shift can
 * carry a letter from one case into the other.
 */
public class Sol {

    private static final String ALPHABET =
            "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";
    private static final int END_OF_INPUT = 0;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        StringBuilder messages = new StringBuilder();
        String line;
        while ((line = reader.readLine()) != null) {
            line = line.trim();
            if (line.isEmpty()) {
                continue;
            }

            int keyCount = Integer.parseInt(line);
            if (keyCount == END_OF_INPUT) {
                break;
            }

            StringTokenizer keyTokens = new StringTokenizer(reader.readLine());
            int[] keys = new int[keyCount];
            for (int i = 0; i < keyCount; i++) {
                keys[i] = Integer.parseInt(keyTokens.nextToken());
            }

            String ciphertext = reader.readLine().trim();
            messages.append(decode(ciphertext, keys)).append('\n');
        }

        System.out.print(messages);
    }

    /** Undoes the encryption by shifting each letter back by its key. */
    private static String decode(String ciphertext, int[] keys) {
        StringBuilder plaintext = new StringBuilder(ciphertext.length());
        for (int position = 0; position < ciphertext.length(); position++) {
            int shift = keys[position % keys.length];
            int enciphered = ALPHABET.indexOf(ciphertext.charAt(position));
            int original = ((enciphered - shift) % ALPHABET.length() + ALPHABET.length())
                    % ALPHABET.length();
            plaintext.append(ALPHABET.charAt(original));
        }
        return plaintext.toString();
    }

MINIMUM_LEG = 1.0


def longest_reachable(limit: float) -> float:
    """Return the greatest total road length within straight-line ``limit``.

    A route is a horizontal leg plus a vertical leg. For an integral number of
    horizontal blocks the vertical leg can stretch to whatever the straight-line
    budget still allows, but never below one block. The unconstrained diagonal
    is also a candidate, so it seeds the search.

        remaining_squared = max(limit * limit - float(horizontal) * horizontal, 0.0)
        vertical = max(MINIMUM_LEG, math.sqrt(remaining_squared))
        best = max(best, horizontal + vertical)

    return best

"""Range Update Query plus Range Sum Query, backed by a lazy segment tree."""

import sys

UPDATE_COMMAND = 0


class LazySegmentTree:
    """Segment tree supporting assign-on-range and sum-over-range."""

    def __init__(self, length: int) -> None:
        self._width = 1
        while self._width < length:
            self._width *= 2

        self._sums = [0] * (2 * self._width)
        self._pending = [None] * (2 * self._width)

    def assign(self, first: int, last: int, value: int) -> None:
        """Set every element in [first, last] to ``value``."""
        self._assign(1, 0, self._width - 1, first, last, value)

    def total(self, first: int, last: int) -> int:
        """Return the sum over [first, last]."""
        return self._total(1, 0, self._width - 1, first, last)

    def _push_down(self, node: int, span: int) -> None:
        """Hand a pending assignment down to both children."""
        value = self._pending[node]
        if value is None:
            return

        half = span // 2
        for child in (2 * node, 2 * node + 1):
            self._sums[child] = value * half
            self._pending[child] = value
        self._pending[node] = None

    def _assign(self, node: int, low: int, high: int, first: int, last: int, value: int) -> None:
        if last < low or high < first:
            return
        if first <= low and high <= last:
            self._sums[node] = value * (high - low + 1)
            self._pending[node] = value
            return

        self._push_down(node, high - low + 1)
        middle = (low + high) // 2
        self._assign(2 * node, low, middle, first, last, value)
        self._assign(2 * node + 1, middle + 1, high, first, last, value)
        self._sums[node] = self._sums[2 * node] + self._sums[2 * node + 1]

    def _total(self, node: int, low: int, high: int, first: int, last: int) -> int:
        if last < low or high < first:
            return 0
        if first <= low and high <= last:
            return self._sums[node]

        self._push_down(node, high - low + 1)
        middle = (low + high) // 2
        return (
            self._total(2 * node, low, middle, first, last)
            + self._total(2 * node + 1, middle + 1, high, first, last)
        )

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;

/**
 * Range Update Query plus Range Sum Query, backed by a lazy segment tree.
 */
public class Sol {

    private static final int UPDATE_COMMAND = 0;

    /** Segment tree supporting assign-on-range and sum-over-range. */
    private static final class LazySegmentTree {
        private final int width;
        private final long[] sums;
        private final long[] pending;
        private final boolean[] hasPending;

        LazySegmentTree(int length) {
            int size = 1;
            while (size < length) {
                size *= 2;
            }
            this.width = size;
            this.sums = new long[2 * size];
            this.pending = new long[2 * size];
            this.hasPending = new boolean[2 * size];
        }

        /** Sets every element in [first, last] to {@code value}. */
        void assign(int first, int last, long value) {
            assign(1, 0, width - 1, first, last, value);
        }

        /** Returns the sum over [first, last]. */
        long total(int first, int last) {
            return total(1, 0, width - 1, first, last);
        }

        /** Hands a pending assignment down to both children. */
        private void pushDown(int node, int span) {
            if (!hasPending[node]) {
                return;
            }

            long value = pending[node];
            int half = span / 2;
            for (int child = 2 * node; child <= 2 * node + 1; child++) {
                sums[child] = value * half;
                pending[child] = value;
                hasPending[child] = true;
            }
            hasPending[node] = false;
        }

        private void assign(int node, int low, int high, int first, int last, long value) {
            if (last < low || high < first) {
                return;
            }
            if (first <= low && high <= last) {
                sums[node] = value * (high - low + 1);
                pending[node] = value;
                hasPending[node] = true;
                return;
            }

            pushDown(node, high - low + 1);
            int middle = (low + high) >>> 1;
            assign(2 * node, low, middle, first, last, value);
            assign(2 * node + 1, middle + 1, high, first, last, value);
            sums[node] = sums[2 * node] + sums[2 * node + 1];
        }

        private long total(int node, int low, int high, int first, int last) {
            if (last < low || high < first) {
                return 0;
            }
            if (first <= low && high <= last) {
                return sums[node];
            }

            pushDown(node, high - low + 1);
            int middle = (low + high) >>> 1;
            return total(2 * node, low, middle, first, last)
                 + total(2 * node + 1, middle + 1, high, first, last);
        }
    }

    public static void main(String[] args) throws IOException {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in)));

        input.nextToken();
        int length = (int) input.nval;
        input.nextToken();
        int queryCount = (int) input.nval;

void pushDown(int node, int nodeLen) {

        long long val = lazy[node];
        int left = 2*node, right = 2*node+1;
        int half = nodeLen / 2;
        sumTree[left] = val * half;
        sumTree[right] = val * half;
        lazy[left] = val; hasLazy[left] = true;
        lazy[right] = val; hasLazy[right] = true;
        hasLazy[node] = false;
    }
}

void update(int node, int nodeL, int nodeR, int s, int t, long long x) {
    if (t < nodeL || nodeR < s) return;
    if (s <= nodeL && nodeR <= t) {
        sumTree[node] = x * (long long)(nodeR - nodeL + 1);
        lazy[node] = x;
        hasLazy[node] = true;
        return;
    }
    pushDown(node, nodeR - nodeL + 1);
    int mid = (nodeL + nodeR) / 2;
    update(2*node, nodeL, mid, s, t, x);

}

#include <iostream>

#include <vector>

namespace {

constexpr int kUpdateCommand = 0;

/// Segment tree supporting assign-on-range and sum-over-range.
class LazySegmentTree {
public:
    explicit LazySegmentTree(int length) {
        width_ = 1;
        while (width_ < length) {
            width_ *= 2;
        }
        sums_.assign(2 * width_, 0);
        pending_.assign(2 * width_, 0);
        hasPending_.assign(2 * width_, 0);
    }

    /// Sets every element in [first, last] to `value`.
    void assign(int first, int last, long long value) {
        assign(1, 0, width_ - 1, first, last, value);
    }

    /// Returns the sum over [first, last].
    long long total(int first, int last) {
        return total(1, 0, width_ - 1, first, last);
    }

private:
    /// Hands a pending assignment down to both children.
    void pushDown(int node, int span) {
        if (!hasPending_[node]) {
            return;
        }

        const long long value = pending_[node];
        const int half = span / 2;
        for (int child = 2 * node; child <= 2 * node + 1; ++child) {
            sums_[child] = value * half;
            pending_[child] = value;
            hasPending_[child] = 1;
        }
        hasPending_[node] = 0;
    }

    void assign(int node, int low, int high, int first, int last, long long value) {
        if (last < low || high < first) {
            return;
        }
        if (first <= low && high <= last) {
            sums_[node] = value * (high - low + 1);
            pending_[node] = value;
            hasPending_[node] = 1;
            return;
        }

        pushDown(node, high - low + 1);
        const int middle = (low + high) / 2;
        assign(2 * node, low, middle, first, last, value);
        assign(2 * node + 1, middle + 1, high, first, last, value);
        sums_[node] = sums_[2 * node] + sums_[2 * node + 1];
    }

    long long total(int node, int low, int high, int first, int last) {
        if (last < low || high < first) {
            return 0;
        }
        if (first <= low && high <= last) {
            return sums_[node];
        }

        pushDown(node, high - low + 1);
        const int middle = (low + high) / 2;
        return total(2 * node, low, middle, first, last)
             + total(2 * node + 1, middle + 1, high, first, last);
    }

    int width_;
    std::vector<long long> sums_;
    std::vector<long long> pending_;
    std::vector<char> hasPending_;
};

}  // namespace

int main() {
    std::ios_base::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int length, queryCount;
    std::cin >> length >> queryCount;

    LazySegmentTree tree(length);
    std::ostringstream answers;

    for (int query = 0; query < queryCount; ++query) {
        int command, first, last;
        std::cin >> command >> first >> last;

        if (command == kUpdateCommand) {
            long long value;
            std::cin >> value;
            tree.assign(first, last, value);
        } else {
            answers << tree.total(first, last) << '\n';
        }
    }

    std::cout << answers.str();
    return 0;
}
