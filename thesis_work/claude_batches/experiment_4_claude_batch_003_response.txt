/**
 * Sum of minimum pairwise distances between points that eventually meet.
 *
 * <p>Two points converge exactly when the left one is not slower than the right
 * one, so we sweep by coordinate and query for smaller-or-equal speeds.
 */
public class Sol {

    /** Binary indexed tree tracking both a count and a sum of coordinates. */
    private static final class FenwickTree {
        private final int size;
        private final long[] counts;
        private final long[] coordinateSums;

        FenwickTree(int size) {
            this.size = size;
            this.counts = new long[size + 1];
            this.coordinateSums = new long[size + 1];
        }

        /** Records one point with the given coordinate at {@code position}. */
        void add(int position, long coordinate) {
            for (int i = position; i <= size; i += i & -i) {
                counts[i]++;
                coordinateSums[i] += coordinate;
            }
        }

        /** Returns {count, coordinate sum} over ranks 1..{@code position}. */
        long[] prefix(int position) {
            long count = 0;
            long coordinateSum = 0;
            for (int i = position; i > 0; i -= i & -i) {
                count += counts[i];
                coordinateSum += coordinateSums[i];
            }
            return new long[] {count, coordinateSum};
        }
    }

    public static void main(String[] args) throws IOException {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in)));

        input.nextToken();
        int pointCount = (int) input.nval;

        long[] coordinates = new long[pointCount];
        long[] velocities = new long[pointCount];
        for (int i = 0; i < pointCount; i++) {
            input.nextToken();
            coordinates[i] = (long) input.nval;
        }
        for (int i = 0; i < pointCount; i++) {
            input.nextToken();
            velocities[i] = (long) input.nval;
        }

        long[] distinctVelocities = distinctSorted(velocities);
        FenwickTree tree = new FenwickTree(distinctVelocities.length);

        Integer[] byCoordinate = new Integer[pointCount];
        Arrays.setAll(byCoordinate, i -> i);
        Arrays.sort(byCoordinate, (a, b) -> Long.compare(coordinates[a], coordinates[b]));

        long totalDistance = 0;
        for (int index : byCoordinate) {
            int rank = rankOf(distinctVelocities, velocities[index]) + 1;
            long[] earlier = tree.prefix(rank);
            totalDistance += earlier[0] * coordinates[index] - earlier[1];
            tree.add(rank, coordinates[index]);
        }

        System.out.println(totalDistance);
    }

    private static long[] distinctSorted(long[] values) {
        long[] sorted = values.clone();
        Arrays.sort(sorted);
        int distinctCount = 0;
        for (int i = 0; i < sorted.length; i++) {
            if (i == 0 || sorted[i] != sorted[i - 1]) {
                sorted[distinctCount++] = sorted[i];
            }
        }
        return Arrays.copyOf(sorted, distinctCount);
    }

    private static int rankOf(long[] sortedDistinct, long value) {
        int low = 0;
        int high = sortedDistinct.length;
        while (low < high) {
            int middle = (low + high) >>> 1;
            if (sortedDistinct[middle] < value) {
                low = middle + 1;
            } else {
                high = middle;
            }
        }
        return low;
    }

"""Size and minimal basis of the span of sparse GF(2) vectors."""

import sys

MODULUS = 10**9 + 7
GROUND = 0  # virtual node standing in for "no second coordinate"


class DisjointSet:
    """Union-find over coordinate nodes, with path compression."""

    def __init__(self, size: int) -> None:
        self._parent = list(range(size + 1))

    def find(self, node: int) -> int:
        root = node
        while self._parent[root] != root:
            root = self._parent[root]
        while self._parent[node] != root:
            self._parent[node], node = root, self._parent[node]
        return root

    def union(self, first: int, second: int) -> bool:
        """Merge the two components; return True when they were distinct."""
        first_root, second_root = self.find(first), self.find(second)
        if first_root == second_root:
            return False
        self._parent[first_root] = second_root
        return True

    /** Union-find over coordinate nodes, with path compression. */
    private static final class DisjointSet {
        private final int[] parent;

        DisjointSet(int size) {
            parent = new int[size + 1];
            for (int node = 0; node <= size; node++) {
                parent[node] = node;
            }
        }

        int find(int node) {
            while (parent[node] != node) {
                parent[node] = parent[parent[node]];
                node = parent[node];
            }
            return node;
        }

        /** Merges the two components; returns true when they were distinct. */
        boolean union(int first, int second) {
            int firstRoot = find(first);
            int secondRoot = find(second);
            if (firstRoot == secondRoot) {
                return false;
            }
            parent[firstRoot] = secondRoot;
            return true;
        }
    }

MODULUS = 10**9 + 7
MAX_OPERATIONS = 200_001

# A digit only matters through how many operations remain once it first
# overflows past 9. `LENGTH_AFTER[t]` is the length that a freshly created
# digit 0 grows to after t further operations: it becomes "1" after 9 more
# steps and splits into "10" after 10 more.
_SPLIT_DELAY_ONE = 9
_SPLIT_DELAY_TWO = 10


def _build_length_table() -> list[int]:
    lengths = [0] * (MAX_OPERATIONS + 1)
    lengths[0] = 1
    for steps in range(1, MAX_OPERATIONS + 1):
        first = 1 if steps <= _SPLIT_DELAY_ONE - 1 else lengths[steps - _SPLIT_DELAY_ONE]
        second = 1 if steps <= _SPLIT_DELAY_TWO - 1 else lengths[steps - _SPLIT_DELAY_TWO]
        lengths[steps] = (first + second) % MODULUS
    return lengths


LENGTH_AFTER = _build_length_table()

    """Length that a single ``digit`` reaches after ``operations`` steps."""

def total_candles_burned(initial_candles: int, stubs_per_candle: int) -> int:
    """Return how many candles burn in total, recycling stubs as we go."""
    burned = initial_candles
    stubs_on_hand = initial_candles

    while stubs_on_hand >= stubs_per_candle:
        recycled = stubs_on_hand // stubs_per_candle
        burned += recycled
        # The recycled candles burn down into stubs of their own.
        stubs_on_hand = stubs_on_hand % stubs_per_candle + recycled

    return burned
