def _build_distinct_prime_counts(limit: int) -> list[int]:
    """Sieve giving, for every n < limit, its number of distinct prime factors."""
    counts = [0] * limit
    for candidate in range(2, limit):
        if counts[candidate] == 0:  # candidate is prime
            for multiple in range(candidate, limit, candidate):
                counts[multiple] += 1
    return counts


DISTINCT_PRIME_COUNTS = _build_distinct_prime_counts(MAX_VALUE)


class Solution:
    def maximumScore(self, nums: list[int], k: int) -> int:
        """Return the maximum score achievable with exactly k operations.

        Each element may be chosen as the "dominant" element of any subarray
        where its prime score is strictly greater than everything to its left
        and greater-or-equal to everything on its right. Counting those
        subarrays tells us how many times an element can be used; we then spend
        the k operations on the largest values first.
        """
        prime_scores = [DISTINCT_PRIME_COUNTS[value] for value in nums]
        reach = self._count_dominated_subarrays(prime_scores)

        score = 1
        operations_left = k
        for index in sorted(range(len(nums)), key=lambda i: nums[i], reverse=True):
            if operations_left == 0:
                break
            uses = min(reach[index], operations_left)
            score = score * pow(nums[index], uses, MODULUS) % MODULUS
            operations_left -= uses

        return score

    @staticmethod
    def _count_dominated_subarrays(prime_scores: list[int]) -> list[int]:
        """For each index, how many subarrays it dominates by prime score."""
        length = len(prime_scores)

        # Nearest index to the left with a strictly greater score.
        previous_greater = [-1] * length
        stack: list[int] = []
        for index in range(length):
            while stack and prime_scores[stack[-1]] < prime_scores[index]:
                stack.pop()
            previous_greater[index] = stack[-1] if stack else -1
            stack.append(index)

        # Nearest index to the right with a greater-or-equal score.
        next_greater_or_equal = [length] * length
        stack.clear()

            while stack and prime_scores[stack[-1]] <= prime_scores[index]:
                stack.pop()
            next_greater_or_equal[index] = stack[-1] if stack else length
            stack.append(index)

        return [
            (index - previous_greater[index]) * (next_greater_or_equal[index] - index)
            for index in range(length)

            stack.push(i);

            right[i] = stack.isEmpty() ? n : stack.peek();

/**
 * Maximum score obtainable by choosing k subarrays greedily by value.
 */
class Solution {

    private static final long MODULUS = 1_000_000_007L;

    /**
     * Returns the maximum score achievable with exactly k operations.
     *
     * <p>Each element may be chosen as the dominant element of any subarray
     * where its prime score beats everything to its left and ties-or-beats
     * everything to its right. We spend the k operations on the largest values
     * first, limited by how many subarrays each element dominates.
     */
    public int maximumScore(List<Integer> nums, int k) {
        int[] primeScores = computePrimeScores(nums);
        long[] reach = countDominatedSubarrays(primeScores);

        Integer[] byValueDescending = new Integer[nums.size()];
        Arrays.setAll(byValueDescending, i -> i);
        Arrays.sort(byValueDescending, (a, b) -> Integer.compare(nums.get(b), nums.get(a)));

        long score = 1;
        long operationsLeft = k;
        for (int index : byValueDescending) {
            if (operationsLeft == 0) {
                break;
            }
            long uses = Math.min(reach[index], operationsLeft);
            score = score * modularPower(nums.get(index), uses) % MODULUS;
            operationsLeft -= uses;
        }

        return (int) score;
    }

    /** Number of distinct prime factors for every value, via a sieve. */
    private int[] computePrimeScores(List<Integer> nums) {
        int limit = 0;
        for (int value : nums) {
            limit = Math.max(limit, value);
        }
        limit++;

        int[] distinctPrimeCounts = new int[limit];
        for (int candidate = 2; candidate < limit; candidate++) {
            if (distinctPrimeCounts[candidate] == 0) { // candidate is prime
                for (int multiple = candidate; multiple < limit; multiple += candidate) {
                    distinctPrimeCounts[multiple]++;
                }
            }
        }

        int[] primeScores = new int[nums.size()];
        for (int i = 0; i < nums.size(); i++) {
            primeScores[i] = distinctPrimeCounts[nums.get(i)];
        }
        return primeScores;
    }

    /** For each index, how many subarrays it dominates by prime score. */
    private long[] countDominatedSubarrays(int[] primeScores) {
        int length = primeScores.length;

        int[] previousGreater = new int[length];
        Deque<Integer> stack = new ArrayDeque<>();
        for (int index = 0; index < length; index++) {
            while (!stack.isEmpty() && primeScores[stack.peek()] < primeScores[index]) {
                stack.pop();
            }
            previousGreater[index] = stack.isEmpty() ? -1 : stack.peek();
            stack.push(index);
        }

        int[] nextGreaterOrEqual = new int[length];
        stack.clear();
        for (int index = length - 1; index >= 0; index--) {
            while (!stack.isEmpty() && primeScores[stack.peek()] <= primeScores[index]) {
                stack.pop();
            }
            nextGreaterOrEqual[index] = stack.isEmpty() ? length : stack.peek();
            stack.push(index);
        }

        long[] reach = new long[length];
        for (int index = 0; index < length; index++) {
            reach[index] = (long) (index - previousGreater[index])
                    * (nextGreaterOrEqual[index] - index);
        }
        return reach;
    }

    /** Computes base^exponent modulo {@link #MODULUS}. */
    private long modularPower(long base, long exponent) {
        long result = 1;
        long factor = base % MODULUS;
        for (long remaining = exponent; remaining > 0; remaining >>= 1) {
            if ((remaining & 1L) == 1L) {
                result = result * factor % MODULUS;
            }
            factor = factor * factor % MODULUS;
        }
        return result;
    }

def smallest_multiple_outside(l: int, r: int, d: int) -> int:
    """Return the least positive multiple of ``d`` that is not within [l, r].

    If ``d`` itself already falls outside the forbidden range it is the answer.
    Otherwise every multiple up to ``r`` is blocked, so the first free one is
    the next multiple after ``r``.

    if d < l or d > r:
        return d
    return (r // d + 1) * d

"""Sum of minimum pairwise distances between points that eventually meet."""

import sys


class FenwickTree:
    """Binary indexed tree tracking both a count and a sum of coordinates."""

    def __init__(self, size: int) -> None:
        self._size = size
        self._counts = [0] * (size + 1)
        self._coordinate_sums = [0] * (size + 1)

    def add(self, position: int, coordinate: int) -> None:
        """Record one point with the given coordinate at ``position``."""
        while position <= self._size:
            self._counts[position] += 1
            self._coordinate_sums[position] += coordinate
            position += position & -position

    def prefix(self, position: int) -> tuple[int, int]:
        """Return (count, coordinate sum) over ranks 1..``position``."""
        count = 0
        coordinate_sum = 0
        while position > 0:
            count += self._counts[position]
            coordinate_sum += self._coordinate_sums[position]
            position -= position & -position
        return count, coordinate_sum


def main() -> None:
    tokens = sys.stdin.buffer.read().split()
    point_count = int(tokens[0])
    coordinates = [int(value) for value in tokens[1 : 1 + point_count]]
    velocities = [int(value) for value in tokens[1 + point_count : 1 + 2 * point_count]]

    # Two points converge exactly when the left one is not slower than the
    # right one, so we sweep by coordinate and query smaller-or-equal speeds.
    velocity_rank = {speed: rank for rank, speed in enumerate(sorted(set(velocities)), start=1)}
    tree = FenwickTree(len(velocity_rank))

    total_distance = 0
    for coordinate, velocity in sorted(zip(coordinates, velocities)):
        earlier_count, earlier_sum = tree.prefix(velocity_rank[velocity])
        total_distance += earlier_count * coordinate - earlier_sum
        tree.add(velocity_rank[velocity], coordinate)

    print(total_distance)


if __name__ == "__main__":
    main()
