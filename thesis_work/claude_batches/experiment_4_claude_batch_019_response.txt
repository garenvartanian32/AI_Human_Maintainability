    std::string cur;
    for (char c : S) {
        if (c == '.' || c == '?' || c == '!') {
            sentences.push_back(cur);
            cur.clear();
        } else {
            cur += c;
        }

        if (trimmed == "I" || (trimmed.size() >= 2 && trimmed[0] == 'I' && trimmed[1] == ' ')) {
            count++;
        }
    }
    return count;
