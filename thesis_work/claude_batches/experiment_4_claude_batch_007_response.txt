def can_split_evenly(unit_weights: list[int]) -> bool:
    """True when some subset of the weights sums to exactly half the total.

    return bool((reachable_subtotals >> (total // 2)) & 1)

public class Sol {

    private static final int WEIGHT_UNIT = 100;

    public static void main(String[] args) throws IOException {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in)));

        input.nextToken();
        int appleCount = (int) input.nval;

        int[] unitWeights = new int[appleCount];
        int total = 0;
        for (int i = 0; i < appleCount; i++) {
            input.nextToken();
            unitWeights[i] = (int) input.nval / WEIGHT_UNIT;
            total += unitWeights[i];
        }

        System.out.println(canSplitEvenly(unitWeights, total) ? "YES" : "NO");
    }

    /** True when some subset of the weights sums to exactly half the total. */
    private static boolean canSplitEvenly(int[] unitWeights, int total) {
        if (total % 2 != 0) {
            return false;
        }

        int half = total / 2;
        boolean[] reachable = new boolean[half + 1];
        reachable[0] = true;

        for (int weight : unitWeights) {
            // Walk downwards so each apple is used at most once.
            for (int subtotal = half; subtotal >= weight; subtotal--) {
                if (reachable[subtotal - weight]) {
                    reachable[subtotal] = true;
                }
            }
        }

        return reachable[half];
    }

from collections import Counter

ALPHABET = "abcdefghijklmnopqrstuvwxyz"


def max_copies(available: Counter, needed: Counter) -> int:
    """How many whole copies of ``needed`` fit inside ``available``."""
    return min(available[letter] // count for letter, count in needed.items())


def main() -> None:
    lines = sys.stdin.read().split("\n")
    source, first_word, second_word = (line.strip() for line in lines[:3])

    available = Counter(source)
    need_first = Counter(first_word)
    need_second = Counter(second_word)

    # Copies of the two words compete for the same letters, so we simply try
    # every feasible count of the first word and take the best total.
    best_total = -1
    best_first_count = 0
    best_second_count = 0

    for first_count in range(max_copies(available, need_first) + 1):
        leftover = Counter(
            {letter: available[letter] - first_count * need_first[letter]
             for letter in available}
        )
        second_count = max_copies(leftover, need_second)

        if first_count + second_count > best_total:
            best_total = first_count + second_count
            best_first_count = first_count
            best_second_count = second_count

    used = Counter()
    for letter, count in need_first.items():
        used[letter] += best_first_count * count
    for letter, count in need_second.items():
        used[letter] += best_second_count * count

    # Any letters not consumed by a copy can be appended in any order.
    leftover_letters = "".join(
        letter * (available[letter] - used[letter]) for letter in sorted(available)
    )

    print(first_word * best_first_count + second_word * best_second_count + leftover_letters)

/**
 * Rearranges a string to hold as many disjoint copies of b and c as possible.
 *
 * <p>Copies of the two words compete for the same letters, so we try every
 * feasible count of the first word and keep the best total.
 */
public class Sol {

    private static final int ALPHABET_SIZE = 26;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String source = reader.readLine().trim();
        String firstWord = reader.readLine().trim();
        String secondWord = reader.readLine().trim();

        int[] available = letterCounts(source);
        int[] needFirst = letterCounts(firstWord);
        int[] needSecond = letterCounts(secondWord);

        int bestTotal = -1;
        int bestFirstCount = 0;
        int bestSecondCount = 0;

        int firstCeiling = maxCopies(available, needFirst, source.length());
        for (int firstCount = 0; firstCount <= firstCeiling; firstCount++) {
            int secondCount = source.length();
            for (int letter = 0; letter < ALPHABET_SIZE; letter++) {
                if (needSecond[letter] > 0) {
                    int leftover = available[letter] - firstCount * needFirst[letter];
                    secondCount = Math.min(secondCount, leftover / needSecond[letter]);
                }
            }

            if (firstCount + secondCount > bestTotal) {
                bestTotal = firstCount + secondCount;
                bestFirstCount = firstCount;
                bestSecondCount = secondCount;
            }
        }

        StringBuilder answer = new StringBuilder();
        answer.append(firstWord.repeat(bestFirstCount));
        answer.append(secondWord.repeat(bestSecondCount));

        // Any letters not consumed by a copy can be appended in any order.
        for (int letter = 0; letter < ALPHABET_SIZE; letter++) {
            int leftover = available[letter]
                    - bestFirstCount * needFirst[letter]
                    - bestSecondCount * needSecond[letter];
            for (int i = 0; i < leftover; i++) {
                answer.append((char) ('a' + letter));
            }
        }

        System.out.println(answer);
    }

    /** How many whole copies of {@code needed} fit inside {@code available}. */
    private static int maxCopies(int[] available, int[] needed, int ceiling) {
        int copies = ceiling;
        for (int letter = 0; letter < ALPHABET_SIZE; letter++) {
            if (needed[letter] > 0) {
                copies = Math.min(copies, available[letter] / needed[letter]);
            }
        }
        return copies;
    }

    private static int[] letterCounts(String word) {
        int[] counts = new int[ALPHABET_SIZE];
        for (int i = 0; i < word.length(); i++) {
            counts[word.charAt(i) - 'a']++;
        }
        return counts;
    }

            copies = std::min(copies, available[letter] / needed[letter]);
        }
    }
    return copies;
}

int main() {
    std::string source, firstWord, secondWord;
    std::getline(std::cin, source);
    std::getline(std::cin, firstWord);
    std::getline(std::cin, secondWord);

    const LetterCounts available = letterCounts(source);
    const LetterCounts needFirst = letterCounts(firstWord);
    const LetterCounts needSecond = letterCounts(secondWord);

    // Copies of the two words compete for the same letters, so we try every
    // feasible count of the first word and keep the best total.
    int bestTotal = -1;
    int bestFirstCount = 0;
    int bestSecondCount = 0;

    const int firstCeiling = maxCopies(available, needFirst, static_cast<int>(source.size()));
    for (int firstCount = 0; firstCount <= firstCeiling; ++firstCount) {
        int secondCount = static_cast<int>(source.size());
        for (int letter = 0; letter < kAlphabetSize; ++letter) {
            if (needSecond[letter] > 0) {
                const int leftover = available[letter] - firstCount * needFirst[letter];
                secondCount = std::min(secondCount, leftover / needSecond[letter]);
            }
        }

        if (firstCount + secondCount > bestTotal) {
            bestTotal = firstCount + secondCount;
            bestFirstCount = firstCount;
            bestSecondCount = secondCount;
        }
    }

    std::string answer;
    for (int i = 0; i < bestFirstCount; ++i) answer += firstWord;
    for (int i = 0; i < bestSecondCount; ++i) answer += secondWord;

    // Any letters not consumed by a copy can be appended in any order.
    for (int letter = 0; letter < kAlphabetSize; ++letter) {
        const int leftover = available[letter]
                           - bestFirstCount * needFirst[letter]
                           - bestSecondCount * needSecond[letter];
        answer += std::string(leftover, static_cast<char>('a' + letter));
    }

    std::cout << answer << '\n';
    return 0;
}
