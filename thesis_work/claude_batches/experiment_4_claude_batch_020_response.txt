def closest_integer(value):
    '''
    Create a function that takes a value (string) representing a number
    and returns the closest integer to it. If the number is equidistant
    from two integers, round it away from zero.

    Examples
    >>> closest_integer("10")
    10
    >>> closest_integer("15.3")
    15

    Note:
    Rounding away from zero means that if the given number is equidistant
    from two integers, the one you should return is the one that is the
    farthest from zero. For example closest_integer("14.5") should
    return 15 and closest_integer("-14.5") should return -15.
    '''
    import math

    number = float(value)

    # Python's round() uses banker's rounding, which would send 14.5 to 14.
    # Nudging by half before flooring (or ceiling, for negatives) instead
    # rounds halves away from zero as the specification requires.
    if number >= 0:
        return math.floor(number + 0.5)
    return math.ceil(number - 0.5)

    public static long closestInteger(String value) {
        // BigDecimal with HALF_UP rounds halves away from zero exactly as the
        // specification asks, and avoids binary floating-point surprises.
        return new BigDecimal(value.trim())
                .setScale(0, RoundingMode.HALF_UP)
                .longValue();
    }

long closest_integer(std::string value) {
    const double number = std::stod(value);

    // std::round breaks ties away from zero, which is precisely the rule the
    // specification describes for values such as 14.5 and -14.5.
    return static_cast<long>(std::round(number));
