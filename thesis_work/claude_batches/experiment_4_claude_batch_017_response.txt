std::string concatenate(std::vector<std::string> strings) {
    // Reserving up front avoids repeated reallocation while appending, and an
    // empty vector naturally yields an empty string.
    std::size_t totalLength = 0;
    for (const std::string& piece : strings) {
        totalLength += piece.size();

        joined += piece;
    }

def triples_sum_to_zero(l: list):
    """
    triples_sum_to_zero takes a list of integers as an input.
    it returns True if there are three distinct elements in the list that
    sum to zero, and False otherwise.

    >>> triples_sum_to_zero([1, 3, 5, 0])
    False

    True
    >>> triples_sum_to_zero([1, 2, 3, 7])
    False
    >>> triples_sum_to_zero([2, 4, -5, 3, 9, 7])
    True
    >>> triples_sum_to_zero([1])
    False
    """
    n = len(l)
    s = sorted(l)
    for i in range(n):
        lo, hi = i + 1, n - 1
        while lo < hi:
            total = s[i] + s[lo] + s[hi]
            if total == 0:
                return True
            elif total < 0:
                lo += 1
            else:
                hi -= 1
    return False

    // >>> isPalindrome(("zbcd"))
    // (false)
    public static boolean isPalindrome(String text) {
        // Walk inwards from both ends; an empty string never enters the loop
        // and so is reported as a palindrome, which is the expected answer.
        int left = 0;
        int right = text.length() - 1;

        while (left < right) {
            if (text.charAt(left) != text.charAt(right)) {
                return false;
            }
            left++;
            right--;
        }

        return true;
    }

// Checks if given string is a palindrome
// >>> is_palindrome((""))
// (true)
// >>> is_palindrome(("aba"))
// (true)
// >>> is_palindrome(("aaaaa"))
// (true)
// >>> is_palindrome(("zbcd"))
// (false)
bool is_palindrome(std::string text) {
    std::string rev = text;
    std::reverse(rev.begin(), rev.end());
    return text == rev;
}
