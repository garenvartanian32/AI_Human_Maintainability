"""Assign each student to the nearest checkpoint by Manhattan distance."""

import sys


def manhattan_distance(first: tuple[int, int], second: tuple[int, int]) -> int:
    return abs(first[0] - second[0]) + abs(first[1] - second[1])


def nearest_checkpoint_number(
    student: tuple[int, int], checkpoints: list[tuple[int, int]]
) -> int:
    """Return the 1-based index of the closest checkpoint.

    Ties are resolved in favour of the smallest index, which falls out of
    scanning left to right and only replacing on a strict improvement.
    """
    best_number = 1
    best_distance = manhattan_distance(student, checkpoints[0])

    for number, checkpoint in enumerate(checkpoints[1:], start=2):
        distance = manhattan_distance(student, checkpoint)
        if distance < best_distance:
            best_distance = distance
            best_number = number

    return best_number


def main() -> None:
    tokens = sys.stdin.buffer.read().split()
    student_count = int(tokens[0])
    checkpoint_count = int(tokens[1])

    cursor = 2
    students = []
    for _ in range(student_count):
        students.append((int(tokens[cursor]), int(tokens[cursor + 1])))
        cursor += 2

    checkpoints = []
    for _ in range(checkpoint_count):
        checkpoints.append((int(tokens[cursor]), int(tokens[cursor + 1])))
        cursor += 2

    answers = [nearest_checkpoint_number(student, checkpoints) for student in students]
    sys.stdout.write("\n".join(map(str, answers)) + "\n")

"""Count completed step-ups and step-downs from a sequence of foot movements."""

import sys

RAISE_SUFFIX = "u"
END_OF_INPUT = 0

# Where the pair of feet currently is.
ON_FLOOR = 0
RISING = 1
ON_STEP = 2
FALLING = 3


def count_completed_moves(movements: list[str]) -> int:
    """Return how many times both feet finished arriving on or off the step."""
    position = ON_FLOOR
    completed = 0

    for movement in movements:
        is_raise = movement[1] == RAISE_SUFFIX

        if position == ON_FLOOR:
            position = RISING
        elif position == RISING:
            if is_raise:
                position = ON_STEP
                completed += 1
            else:
                position = ON_FLOOR
        elif position == ON_STEP:
            position = FALLING
        else:  # FALLING
            if is_raise:
                position = ON_STEP
            else:
                position = ON_FLOOR
                completed += 1

    return completed


def main() -> None:
    tokens = sys.stdin.buffer.read().split()
    cursor = 0
    answers = []

    while cursor < len(tokens):
        movement_count = int(tokens[cursor])
        cursor += 1
        if movement_count == END_OF_INPUT:
            break

        movements = [tokens[cursor + i].decode() for i in range(movement_count)]
        cursor += movement_count
        answers.append(count_completed_moves(movements))

    sys.stdout.write("\n".join(map(str, answers)) + "\n")

    private static final int END_OF_INPUT = 0;

    // Where the pair of feet currently is.
    private static final int ON_FLOOR = 0;
    private static final int RISING = 1;
    private static final int ON_STEP = 2;
    private static final int FALLING = 3;

    public static void main(String[] args) throws IOException {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in)));

            input.nextToken();
            int movementCount = (int) input.nval;

                break;

            int completed = 0;

                            position = ON_STEP;
                        } else {
                            position = ON_FLOOR;
                            completed++;
                        }
                        break;
                }
            }

            answers.append(completed).append('\n');
        }

import sys


def selection_sort(values: list[int]) -> int:
    """Sort ``values`` in place; return how many real swaps were performed."""
    swap_count = 0

    for boundary in range(len(values)):
        smallest = boundary
        for candidate in range(boundary + 1, len(values)):
            if values[candidate] < values[smallest]:
                smallest = candidate

        # Only count a swap when the element actually moves.
        if smallest != boundary:
            values[boundary], values[smallest] = values[smallest], values[boundary]
            swap_count += 1

    return swap_count

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;

/**
 * Smallest candidate total that beats every rival and every existing score.
 *
 * <p>The answer must be at least the current maximum, and large enough that
 * {@code count} copies of it strictly exceed twice the current total.
 */
public class Sol {

    public static void main(String[] args) throws IOException {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in)));

        input.nextToken();
        int count = (int) input.nval;

        long total = 0;
        long largestScore = 0;
        for (int i = 0; i < count; i++) {
            input.nextToken();
            long score = (long) input.nval;
            total += score;
            largestScore = Math.max(largestScore, score);
        }

        long smallestWinning = largestScore;
        while (count * smallestWinning <= 2 * total) {
            smallestWinning++;
        }

        System.out.println(smallestWinning);
    }
}
