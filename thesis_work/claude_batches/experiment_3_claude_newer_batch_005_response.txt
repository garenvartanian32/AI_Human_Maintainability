public class Sol {
    static String rebuild(String record) {
        Map<Character, Character> behind = new HashMap<>();
        Map<Character, Character> ahead = new HashMap<>();
        Set<Character> seen = new LinkedHashSet<>();

        char current = record.charAt(0);
        seen.add(current);
        int i = 1;
        while (i < record.length()) {
            boolean backwards = false;
            if (record.charAt(i) == '-') {
                backwards = true;
                i += 2;
            }
            char next = record.charAt(i);
            i++;
            seen.add(next);
            if (backwards) {
                behind.put(current, next);
                ahead.put(next, current);
            } else {
                ahead.put(current, next);
                behind.put(next, current);
            }
            current = next;
        }

        char head = 0;
        for (char car : seen) {
            if (!ahead.containsKey(car)) { head = car; break; }
        }

        StringBuilder sb = new StringBuilder();
        Character car = head;
        while (car != null) {
            sb.append((char) car);
            car = behind.get(car);
        }
        return sb.toString();
    }

        for (int i = 0; i < n; i++) {

            seen.push_back(next);

            behind[current] = next;
            ahead[next] = current;
        } else {
            ahead[current] = next;
            behind[next] = current;
        }
        current = next;
    }

    char head = 0;

        if (!ahead.count(car)) { head = car; break; }
