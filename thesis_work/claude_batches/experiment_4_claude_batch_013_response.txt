"""Cheapest recolouring of a grid so that colours depend on (row+col) mod 3."""

import sys
from itertools import permutations

GROUP_COUNT = 3


def main() -> None:
    tokens = sys.stdin.buffer.read().split()
    grid_size = int(tokens[0])
    colour_count = int(tokens[1])
    cursor = 2

    repaint_cost = []
    for _ in range(colour_count):
        repaint_cost.append([int(token) for token in tokens[cursor : cursor + colour_count]])
        cursor += colour_count

    # Cells sharing (row + column) mod 3 must end up the same colour, so only
    # the tally of current colours per group matters.
    tally = [[0] * colour_count for _ in range(GROUP_COUNT)]
    for row in range(grid_size):
        for column in range(grid_size):
            current_colour = int(tokens[cursor]) - 1
            cursor += 1
            tally[(row + column) % GROUP_COUNT][current_colour] += 1

    # Cost of turning every cell of a group into one target colour.
    group_cost = [
        [
            sum(tally[group][source] * repaint_cost[source][target]
                for source in range(colour_count))
            for target in range(colour_count)
        ]
        for group in range(GROUP_COUNT)
    ]

    # The three groups must receive three distinct colours.
    print(min(
        group_cost[0][first] + group_cost[1][second] + group_cost[2][third]
        for first, second, third in permutations(range(colour_count), GROUP_COUNT)
    ))

/**
 * Cheapest recolouring of a grid so that colours depend on (row+col) mod 3.
 *
 * <p>Cells sharing {@code (row + column) % 3} must end up the same colour, so
 * only the tally of current colours per group matters. The three groups must
 * then receive three distinct colours.
 */
public class Sol {

    private static final int GROUP_COUNT = 3;

    public static void main(String[] args) throws IOException {
        StreamTokenizer input = new StreamTokenizer(
                new BufferedReader(new InputStreamReader(System.in)));

        input.nextToken();
        int gridSize = (int) input.nval;
        input.nextToken();
        int colourCount = (int) input.nval;

        long[][] repaintCost = new long[colourCount][colourCount];
        for (int source = 0; source < colourCount; source++) {
            for (int target = 0; target < colourCount; target++) {
                input.nextToken();
                repaintCost[source][target] = (long) input.nval;
            }
        }

        long[][] tally = new long[GROUP_COUNT][colourCount];
        for (int row = 0; row < gridSize; row++) {
            for (int column = 0; column < gridSize; column++) {
                input.nextToken();
                int currentColour = (int) input.nval - 1;
                tally[(row + column) % GROUP_COUNT][currentColour]++;
            }
        }

        long[][] groupCost = new long[GROUP_COUNT][colourCount];
        for (int group = 0; group < GROUP_COUNT; group++) {
            for (int target = 0; target < colourCount; target++) {
                long cost = 0;
                for (int source = 0; source < colourCount; source++) {
                    cost += tally[group][source] * repaintCost[source][target];
                }
                groupCost[group][target] = cost;
            }
        }

        long best = Long.MAX_VALUE;
        for (int first = 0; first < colourCount; first++) {
            for (int second = 0; second < colourCount; second++) {
                if (second == first) {
                    continue;
                }
                for (int third = 0; third < colourCount; third++) {
                    if (third == first || third == second) {
                        continue;
                    }
                    best = Math.min(best,
                            groupCost[0][first] + groupCost[1][second] + groupCost[2][third]);
                }
            }
        }

        System.out.println(best);
    }

def main():
    data = sys.stdin.read().split()
    idx = 0
    x1 = int(data[idx]); y1 = int(data[idx+1]); x2 = int(data[idx+2]); y2 = int(data[idx+3]); idx += 4
    n = int(data[idx]); idx += 1
    fountains = []
    for _ in range(n):
        X = int(data[idx]); Y = int(data[idx+1]); idx += 2
        fountains.append((X, Y))

    r = 10.0
    dx = abs(x1 - x2)
    dy = abs(y1 - y2)
    manhattan = (dx + dy) * 100.0

    if x1 == x2 or y1 == y2:
        # straight line; check blocking fountain
        penalty = 0.0
        if y1 == y2:
            lo, hi = min(x1, x2), max(x1, x2)
            for X, Y in fountains:
                if Y == y1 and lo < X < hi:
                    penalty = r * (math.pi - 2)
                    break
        else:
            lo, hi = min(y1, y2), max(y1, y2)
            for X, Y in fountains:
                if X == x1 and lo < Y < hi:
                    penalty = r * (math.pi - 2)
                    break
        print(f"{manhattan + penalty:.15f}")
        return

    xlo, xhi = min(x1, x2), max(x1, x2)
    ylo, yhi = min(y1, y2), max(y1, y2)
    xdir = 1 if x2 > x1 else -1
    ydir = 1 if y2 > y1 else -1

    candidates = []
    for X, Y in fountains:
        if xlo < X < xhi and ylo < Y < yhi:
            candidates.append((X * xdir, Y * ydir))

    candidates.sort()
    # longest strictly increasing subsequence on the Y-transformed value
    tails = []
    for _, ty in candidates:
        pos = bisect.bisect_left(tails, ty)
        if pos == len(tails):
            tails.append(ty)
        else:
            tails[pos] = ty
    lis_len = len(tails)

    bonus = r * (2 - math.pi/2)
    answer = manhattan - lis_len * bonus
    print(f"{answer:.15f}")

"""Shortest walk on a city grid, allowing corner cuts around fountains."""

import sys
from bisect import bisect_left
from math import pi, sqrt

BLOCK_LENGTH = 100.0
FOUNTAIN_RADIUS = 10.0

# Cutting a corner around a fountain replaces two quarter-block legs with a
# quarter arc, saving this much; a fountain sitting directly on a straight
# stretch instead forces a detour of this much.
CORNER_SAVING = FOUNTAIN_RADIUS * (2 - pi / 2)
STRAIGHT_DETOUR = FOUNTAIN_RADIUS * (pi - 2)


def longest_increasing_run(points: list[tuple[int, int]]) -> int:
    """Length of the longest strictly increasing chain by the second coordinate.

    Corner cuts can only be chained along a monotone staircase, so the number of
    usable fountains is exactly this chain length.
    """
    tails: list[int] = []
    for _, second in points:
        slot = bisect_left(tails, second)
        if slot == len(tails):
            tails.append(second)
        else:
            tails[slot] = second
    return len(tails)


def main() -> None:
    tokens = sys.stdin.buffer.read().split()
    start_x, start_y, end_x, end_y = (int(tokens[index]) for index in range(4))
    fountain_count = int(tokens[4])

    fountains = []
    cursor = 5
    for _ in range(fountain_count):
        fountains.append((int(tokens[cursor]), int(tokens[cursor + 1])))
        cursor += 2

    walk = (abs(start_x - end_x) + abs(start_y - end_y)) * BLOCK_LENGTH

    # A purely horizontal or vertical trip has no corners to cut; a fountain in
    # the way only costs extra.
    if start_x == end_x or start_y == end_y:
        blocked = _straight_path_blocked(start_x, start_y, end_x, end_y, fountains)
        print(f"{walk + (STRAIGHT_DETOUR if blocked else 0.0):.15f}")
        return

    # Reflect coordinates so the trip always runs in the increasing direction,
    # which turns "usable fountains" into a plain increasing subsequence.
    x_sign = 1 if end_x > start_x else -1
    y_sign = 1 if end_y > start_y else -1
    low_x, high_x = sorted((start_x, end_x))
    low_y, high_y = sorted((start_y, end_y))

    inside = sorted(
        (x * x_sign, y * y_sign)
        for x, y in fountains
        if low_x < x < high_x and low_y < y < high_y
    )

    print(f"{walk - longest_increasing_run(inside) * CORNER_SAVING:.15f}")


def _straight_path_blocked(
    start_x: int, start_y: int, end_x: int, end_y: int, fountains: list[tuple[int, int]]
) -> bool:
    if start_y == end_y:
        low, high = sorted((start_x, end_x))
        return any(y == start_y and low < x < high for x, y in fountains)

    low, high = sorted((start_y, end_y))
    return any(x == start_x and low < y < high for x, y in fountains)


if __name__ == "__main__":
    main()

                } else {

        }
        return length;

/// Longest strictly increasing chain by the second coordinate.
int longestIncreasingRun(const std::vector<std::pair<long long, long long>>& points) {
    std::vector<long long> tails;
    for (const auto& point : points) {
        auto slot = std::lower_bound(tails.begin(), tails.end(), point.second);
        if (slot == tails.end()) {
            tails.push_back(point.second);
        } else {
            *slot = point.second;
        }
    }
    return static_cast<int>(tails.size());
}
