        int total = 0;
        for (int i = 0; i < n; i++) {
            in.nextToken();
            units[i] = (int) in.nval / 100;
            total += units[i];
        }

        if (total % 2 != 0) {
            System.out.println("NO");
            return;
        }

        BitSet reachable = new BitSet(total + 1);
        reachable.set(0);
        for (int u : units) {
            BitSet shifted = new BitSet(total + 1);
            for (int s = reachable.nextSetBit(0); s >= 0; s = reachable.nextSetBit(s + 1)) {
                if (s + u <= total) shifted.set(s + u);
            }
            reachable.or(shifted);
        }
        System.out.println(reachable.get(total / 2) ? "YES" : "NO");
    }

def main():
    lines = sys.stdin.read().split('\n')
    a, b, c = lines[0].strip(), lines[1].strip(), lines[2].strip()

    have = Counter(a)
    need_b = Counter(b)
    need_c = Counter(c)

    max_b = min(have[ch] // need_b[ch] for ch in need_b)

    best = (-1, 0, 0)
    for copies_b in range(max_b + 1):
        left = {ch: have[ch] - copies_b * need_b[ch] for ch in have}
        copies_c = min((left.get(ch, 0) // need_c[ch] for ch in need_c), default=0)
        if copies_b + copies_c > best[0]:
            best = (copies_b + copies_c, copies_b, copies_c)

    _, copies_b, copies_c = best
    used = Counter()
    for ch in need_b:
        used[ch] += copies_b * need_b[ch]
    for ch in need_c:
        used[ch] += copies_c * need_c[ch]

    tail = ''.join(ch * (have[ch] - used[ch]) for ch in sorted(have))
    print(b * copies_b + c * copies_c + tail)

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0]); m = int(data[1])
    p = [int(x) for x in data[2:2 + n]]

    parent = list(range(n + 1))

    def find(x):
        root = x
        while parent[root] != root:
            root = parent[root]
        while parent[x] != root:
            parent[x], x = root, parent[x]
        return root

    pos = 2 + n
    for _ in range(m):
        u = find(int(data[pos])); v = find(int(data[pos + 1]))
        pos += 2
        if u != v:
            parent[u] = v

    buckets = {}
    for i in range(1, n + 1):
        buckets.setdefault(find(i), []).append(i)

    answer = [0] * (n + 1)
    for slots in buckets.values():
        values = sorted((p[i - 1] for i in slots), reverse=True)
        for slot, value in zip(slots, values):
            answer[slot] = value

    print(' '.join(map(str, answer[1:])))

import java.io.*;
import java.util.*;

public class Sol {
    static int[] parent;

    static int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }

    public static void main(String[] args) throws IOException {
        DataInputStream in = new DataInputStream(new BufferedInputStream(System.in, 1 << 16));
        int n = nextInt(in), m = nextInt(in);
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) p[i] = nextInt(in);

        parent = new int[n + 1];
        for (int i = 0; i <= n; i++) parent[i] = i;
        for (int e = 0; e < m; e++) {
            int u = find(nextInt(in)), v = find(nextInt(in));
            if (u != v) parent[u] = v;
        }

        List<List<Integer>> buckets = new ArrayList<>();
        int[] bucketOf = new int[n + 1];
        Arrays.fill(bucketOf, -1);
        for (int i = 1; i <= n; i++) {
            int root = find(i);
            if (bucketOf[root] < 0) {
                bucketOf[root] = buckets.size();
                buckets.add(new ArrayList<>());
            }
            buckets.get(bucketOf[root]).add(i);
        }

        int[] answer = new int[n + 1];
        for (List<Integer> slots : buckets) {
            List<Integer> values = new ArrayList<>();
            for (int slot : slots) values.add(p[slot]);
            values.sort(Collections.reverseOrder());
            for (int i = 0; i < slots.size(); i++) {
                answer[slots.get(i)] = values.get(i);
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            if (i > 1) sb.append(' ');
            sb.append(answer[i]);
        }
        System.out.println(sb);
    }

    private static int nextInt(DataInputStream in) throws IOException {
        int value = 0, b = in.read();
        while (b < '0' || b > '9') b = in.read();
        while (b >= '0' && b <= '9') {
            value = value * 10 + (b - '0');
            b = in.read();
        }
        return value;
    }
}

import sys


def rebuild(record):
    behind = {}
    ahead = {}
    seen = set()

    current = record[0]
    seen.add(current)
    i = 1
    while i < len(record):
        if record[i] == '-':
            backwards = True
            i += 2
        else:
            backwards = False
        nxt = record[i]
        i += 1
        seen.add(nxt)
        if backwards:
            behind[current] = nxt
            ahead[nxt] = current
        else:
            ahead[current] = nxt
            behind[nxt] = current
        current = nxt

    head = next(car for car in seen if car not in ahead)
    order = []
    car = head
    while car is not None:
        order.append(car)
        car = behind.get(car)
    return ''.join(order)


def main():
    lines = sys.stdin.read().split('\n')
    n = int(lines[0].strip())
    out = [rebuild(lines[i].strip()) for i in range(1, n + 1)]
    sys.stdout.write('\n'.join(out) + '\n')
