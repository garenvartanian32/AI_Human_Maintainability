import sys

def main():
    data = sys.stdin.read().split('\n')
    idx = 0
    m, n = map(int, data[idx].split()); idx += 1
    free = [(1, m)]  # list of (start, length) free blocks, in order
    blocks = {}  # id -> (start, length) or None if erased
    next_id = 1
    out = []
    for _ in range(n):
        parts = data[idx].split(); idx += 1
        if parts[0] == 'alloc':
            need = int(parts[1])
            placed = False
            for i, (s, l) in enumerate(free):
                if l >= need:
                    blocks[next_id] = (s, need)
                    if l == need:
                        free.pop(i)
                    else:
                        free[i] = (s+need, l-need)
                    out.append(str(next_id))
                    next_id += 1
                    placed = True
                    break
            if not placed:
                out.append("NULL")
                next_id += 1
        elif parts[0] == 'erase':
            x = int(parts[1])
            if x in blocks and blocks[x] is not None:
                s, l = blocks[x]
                free.append((s, l))
                free.sort()
                merged = []
                for seg in free:
                    if merged and merged[-1][0]+merged[-1][1] == seg[0]:
                        merged[-1] = (merged[-1][0], merged[-1][1]+seg[1])
                    else:
                        merged.append(seg)
                free = merged
                blocks[x] = None
            else:
                out.append("ILLEGAL_ERASE_ARGUMENT")
        else:  # defragment
            occupied = sorted((s,l,bid) for bid,(s,l) in blocks.items() if blocks[bid] is not None)
            pos = 1
            newblocks = {}
            for s,l,bid in occupied:
                newblocks[bid] = (pos, l)
                pos += l
            blocks = newblocks
            free = [(pos, m-pos+1)] if pos <= m else []
    print('\n'.join(out))

import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st0 = new StringTokenizer(br.readLine());
        int m = Integer.parseInt(st0.nextToken());
        int n = Integer.parseInt(st0.nextToken());
        List<int[]> free = new ArrayList<int[]>();
        free.add(new int[]{1, m});
        Map<Integer, int[]> blocks = new HashMap<Integer, int[]>();
        int nextId = 1;
        StringBuilder out = new StringBuilder();
        for (int q = 0; q < n; q++) {
            String line = br.readLine();
            StringTokenizer st = new StringTokenizer(line);
            String cmd = st.nextToken();
            if (cmd.equals("alloc")) {
                int need = Integer.parseInt(st.nextToken());
                boolean placed = false;
                for (int i = 0; i < free.size(); i++) {
                    int[] f = free.get(i);
                    if (f[1] >= need) {
                        blocks.put(nextId, new int[]{f[0], need});
                        if (f[1] == need) free.remove(i);
                        else free.set(i, new int[]{f[0]+need, f[1]-need});
                        out.append(nextId).append("\n");
                        nextId++;
                        placed = true;
                        break;
                    }
                }
                if (!placed) { out.append("NULL\n"); nextId++; }
            } else if (cmd.equals("erase")) {
                int x = Integer.parseInt(st.nextToken());
                if (blocks.containsKey(x)) {
                    int[] b = blocks.remove(x);
                    free.add(b);
                    free.sort((a,c) -> a[0]-c[0]);
                    List<int[]> merged = new ArrayList<int[]>();
                    for (int[] seg : free) {
                        if (!merged.isEmpty() && merged.get(merged.size()-1)[0]+merged.get(merged.size()-1)[1] == seg[0]) {
                            merged.get(merged.size()-1)[1] += seg[1];
                        } else merged.add(seg);
                    }
                    free = merged;
                } else {
                    out.append("ILLEGAL_ERASE_ARGUMENT\n");
                }
            } else {
                List<Map.Entry<Integer,int[]>> occ = new ArrayList<Map.Entry<Integer,int[]>>(blocks.entrySet());
                occ.sort((a,b) -> a.getValue()[0]-b.getValue()[0]);
                int pos = 1;
                Map<Integer,int[]> newBlocks = new HashMap<Integer,int[]>();
                for (Map.Entry<Integer,int[]> e : occ) {
                    int l = e.getValue()[1];
                    newBlocks.put(e.getKey(), new int[]{pos, l});
                    pos += l;
                }
                blocks = newBlocks;
                free = new ArrayList<int[]>();
                if (pos <= m) free.add(new int[]{pos, m-pos+1});
            }
        }
        System.out.print(out);
    }
}

            int x;
            iss >> x;
            if (blocks.count(x)) {
                free_.push_back(blocks[x]);
                blocks.erase(x);
                sort(free_.begin(), free_.end());
                vector<pair<int,int>> merged;
                for (auto &seg : free_) {
                    if (!merged.empty() && merged.back().first+merged.back().second == seg.first) {
                        merged.back().second += seg.second;
                    } else merged.push_back(seg);
                }
                free_ = merged;
            } else {
                out += "ILLEGAL_ERASE_ARGUMENT\n";
            }
        } else {
            vector<pair<int,pair<int,int>>> occ;
            for (auto &p : blocks) occ.push_back({p.second.first, {p.first, p.second.second}});
            sort(occ.begin(), occ.end());
            int pos = 1;
            map<int,pair<int,int>> newBlocks;
            for (auto &o : occ) {
                int bid = o.second.first, l = o.second.second;
                newBlocks[bid] = {pos, l};
                pos += l;
            }
            blocks = newBlocks;
            free_.clear();
            if (pos <= m) free_.push_back({pos, m-pos+1});
        }
    }
    cout << out;
    return 0;
}
