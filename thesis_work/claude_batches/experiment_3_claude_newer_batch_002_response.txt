public class Sol {
    static int size;
    static long[] cnt;
    static long[] tot;

    static void add(int i, long x) {
        while (i <= size) {
            cnt[i]++;
            tot[i] += x;
            i += i & -i;
        }
    }

    static long[] query(int i) {
        long c = 0, t = 0;
        while (i > 0) {
            c += cnt[i];
            t += tot[i];
            i -= i & -i;
        }
        return new long[]{c, t};
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StreamTokenizer in = new StreamTokenizer(br);
        in.nextToken();
        int n = (int) in.nval;
        long[] xs = new long[n];
        long[] vs = new long[n];
        for (int i = 0; i < n; i++) { in.nextToken(); xs[i] = (long) in.nval; }
        for (int i = 0; i < n; i++) { in.nextToken(); vs[i] = (long) in.nval; }

        long[] sortedV = vs.clone();
        Arrays.sort(sortedV);
        int distinct = 0;
        for (int i = 0; i < n; i++) {
            if (i == 0 || sortedV[i] != sortedV[i - 1]) {
                sortedV[distinct++] = sortedV[i];
            }
        }

        size = distinct;
        cnt = new long[size + 1];
        tot = new long[size + 1];

        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) order[i] = i;
        Arrays.sort(order, (a, b) -> Long.compare(xs[a], xs[b]));

        long total = 0;

            int rank = lowerBound(sortedV, distinct, vs[idx]) + 1;
            long[] res = query(rank);
            total += res[0] * xs[idx] - res[1];
            add(rank, xs[idx]);
        }
        System.out.println(total);
    }

    static int lowerBound(long[] arr, int len, long key) {
        int lo = 0, hi = len;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (arr[mid] < key) lo = mid + 1;
            else hi = mid;
        }
        return lo;
    }

struct Fenwick {
    int size;
    vector<long long> cnt, tot;
    Fenwick(int n) : size(n), cnt(n + 1, 0), tot(n + 1, 0) {}

    void add(int i, long long x) {
        for (; i <= size; i += i & -i) {
            cnt[i]++;
            tot[i] += x;
        }
    }

    pair<long long, long long> query(int i) const {
        long long c = 0, t = 0;
        for (; i > 0; i -= i & -i) {
            c += cnt[i];
            t += tot[i];
        }
        return {c, t};
    }
};

            else:

                live.add(next_id)

            packed = [b for b in memory if b != 0]
            memory = packed + [0] * (m - len(packed))

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer header = new StringTokenizer(br.readLine());
        int t = Integer.parseInt(header.nextToken());
        int m = Integer.parseInt(header.nextToken());

        int[] memory = new int[m];
        Set<Integer> live = new HashSet<>();
        int nextId = 1;
        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < t; i++) {
            StringTokenizer tk = new StringTokenizer(br.readLine());
            String cmd = tk.nextToken();

            if (cmd.equals("alloc")) {
                int need = Integer.parseInt(tk.nextToken());
                int start = -1, run = 0;
                for (int j = 0; j < m; j++) {
                    run = (memory[j] == 0) ? run + 1 : 0;
                    if (run == need) { start = j - need + 1; break; }
                }
                if (start < 0) {
                    sb.append("NULL").append('\n');
                } else {
                    Arrays.fill(memory, start, start + need, nextId);
                    live.add(nextId);
                    sb.append(nextId).append('\n');
                    nextId++;
                }
            } else if (cmd.equals("erase")) {
                int target = Integer.parseInt(tk.nextToken());

                        if (memory[j] == target) memory[j] = 0;
                    }
                } else {
                    sb.append("ILLEGAL_ERASE_ARGUMENT").append('\n');
                }
            } else if (cmd.equals("defragment")) {
                int write = 0;
                for (int j = 0; j < m; j++) {
                    if (memory[j] != 0) memory[write++] = memory[j];
                }
                Arrays.fill(memory, write, m, 0);
            }

                out << nextId << '\n';
                nextId++;
            }
        } else if (cmd == "erase") {
            int target;
            iss >> target;
            if (live.erase(target)) {
                for (int j = 0; j < m; j++) {
                    if (memory[j] == target) memory[j] = 0;
                }
            } else {
                out << "ILLEGAL_ERASE_ARGUMENT\n";
            }
        } else if (cmd == "defragment") {
            int write = 0;
            for (int j = 0; j < m; j++) {
                if (memory[j] != 0) memory[write++] = memory[j];
            }
            fill(memory.begin() + write, memory.end(), 0);
        }
    }
    cout << out.str();
    return 0;
}
