def factorize(q):
    factors = []
    d = 2
    while d * d <= q:
        while q % d == 0:
            factors.append(d)

    return factors

    unordered_map<long long, int> seen;
    seen[start] = 0;
    queue<long long> q;
    q.push(start);

    while (!q.empty()) {
        long long cur = q.front();
        q.pop();
        if (cur == goal) {
            cout << seen[cur] << '\n';
            return 0;
        }
        long long r = cur / SHIFT;
        long long c = cur - r * SHIFT;
        int step = seen[cur] + 1;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                if (dr == 0 && dc == 0) continue;
                long long nxt = (r + dr) * SHIFT + (c + dc);
                if (allowed.count(nxt) && !seen.count(nxt)) {
                    seen[nxt] = step;
                    q.push(nxt);
                }
            }
        }
    }
    cout << -1 << '\n';
    return 0;
}
