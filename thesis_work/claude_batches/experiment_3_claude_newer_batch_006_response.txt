import sys


def main():
    data = sys.stdin.buffer.read().split()
    pos = 0
    t = int(data[pos]); pos += 1
    out = []

    for _ in range(t):
        n = int(data[pos]); pos += 1
        a = [int(x) for x in data[pos:pos + n]]; pos += n
        s = data[pos].decode(); pos += 1

        basis = [0] * 64
        winner = 0
        for i in range(n - 1, -1, -1):
            v = a[i]
            for bit in range(63, -1, -1):
                if not (v >> bit) & 1:
                    continue
                if basis[bit] == 0:
                    basis[bit] = v
                    v = 0
                    if s[i] == '1':
                        winner = 1
                    break
                v ^= basis[bit]
            if winner:
                break
        out.append(winner)

    sys.stdout.write('\n'.join(map(str, out)) + '\n')

public class Sol {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());

        List<String> words = new ArrayList<>();
        words.add("");
        for (int step = 0; step < n; step++) {
            List<String> grown = new ArrayList<>();
            for (String word : words) {
                char top = 'a' - 1;
                for (int i = 0; i < word.length(); i++) {
                    if (word.charAt(i) > top) top = word.charAt(i);
                }
                for (char c = 'a'; c <= top + 1; c++) {
                    grown.add(word + c);
                }
            }
            words = grown;
        }

        StringBuilder sb = new StringBuilder();
        for (String word : words) sb.append(word).append('\n');
        System.out.print(sb);
    }
}

                expire[edge] += need * a

import java.util.*;

public class Sol {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StreamTokenizer in = new StreamTokenizer(br);
        in.nextToken(); int n = (int) in.nval;
        in.nextToken(); long d = (long) in.nval;
        in.nextToken(); long a = (long) in.nval;

        long[][] monsters = new long[n][2];
        for (int i = 0; i < n; i++) {
            in.nextToken(); monsters[i][0] = (long) in.nval;
            in.nextToken(); monsters[i][1] = (long) in.nval;
        }
        Arrays.sort(monsters, (p, q) -> Long.compare(p[0], q[0]));

        long[] xs = new long[n];
        for (int i = 0; i < n; i++) xs[i] = monsters[i][0];

        long[] expire = new long[n + 1];
        long damage = 0, bombs = 0;
        for (int i = 0; i < n; i++) {
            damage -= expire[i];
            long left = monsters[i][1] - damage;
            if (left > 0) {
                long need = (left + a - 1) / a;
                bombs += need;
                damage += need * a;
                int edge = upperBound(xs, monsters[i][0] + 2 * d);
                if (edge <= n) expire[edge] += need * a;
            }
        }
        System.out.println(bombs);
    }

    static int upperBound(long[] arr, long key) {
        int lo = 0, hi = arr.length;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (arr[mid] <= key) lo = mid + 1;
            else hi = mid;
        }
        return lo;
    }
}
