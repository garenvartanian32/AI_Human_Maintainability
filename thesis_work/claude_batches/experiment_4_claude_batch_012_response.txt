namespace {

constexpr char kFirstLetter = 'a';

int gLength;
std::string gBuffer;
std::string gResults;

/**
 * Extends the buffer at `position`, given how many distinct letters are in use.
 *
 * Each letter may be at most one past the largest used so far, which keeps
 * exactly one representative of every partition of the positions and emits the
 * strings already sorted.
 */
void extend(int position, int lettersUsed) {
    if (position == gLength) {
        gResults += gBuffer;
        gResults += '\n';
        return;
    }

    for (int letterIndex = 0; letterIndex <= lettersUsed; ++letterIndex) {
        gBuffer[position] = static_cast<char>(kFirstLetter + letterIndex);
        const int nextUsed = (letterIndex == lettersUsed) ? lettersUsed + 1 : lettersUsed;
        extend(position + 1, nextUsed);
    }
}

}  // namespace

int main() {
    std::cin >> gLength;
    gBuffer.assign(gLength, kFirstLetter);

    extend(0, 0);
    std::cout << gResults;
    return 0;
}

import sys
from bisect import bisect_right


def main() -> None:
    tokens = sys.stdin.buffer.read().split()
    monster_count = int(tokens[0])
    blast_radius = int(tokens[1])
    blast_damage = int(tokens[2])

    monsters = []
    cursor = 3
    for _ in range(monster_count):
        monsters.append((int(tokens[cursor]), int(tokens[cursor + 1])))
        cursor += 2
    monsters.sort()

    positions = [position for position, _ in monsters]

    # Sweeping left to right, every monster must die at its own turn, so bombs
    # are always centred as far right as still covers it. `expiring` records
    # where each blast stops contributing, letting us carry a running total.
    expiring = [0] * (monster_count + 1)
    damage_in_effect = 0
    bombs_used = 0

    for index, (position, health) in enumerate(monsters):
        damage_in_effect -= expiring[index]

        remaining_health = health - damage_in_effect
        if remaining_health <= 0:
            continue

        bombs = -(-remaining_health // blast_damage)
        bombs_used += bombs
        damage_in_effect += bombs * blast_damage

        last_covered = bisect_right(positions, position + 2 * blast_radius)
        if last_covered <= monster_count:
            expiring[last_covered] += bombs * blast_damage

    print(bombs_used)

            positions[i] = monsters[i][0];

                expiring[lastCovered] += bombs * blastDamage;
            }
        }

    private static int upperBound(long[] positions, long key) {

        int high = positions.length;

                low = middle + 1;
            } else {
                high = middle;
            }

                if (candidate >= 1 && n / candidate == n % candidate) {
                    total += candidate;
                }
            }
        }

        System.out.println(total);
    }

            value = nextTerm(value);
            index++;

#include <iostream>
#include <unordered_set>

namespace {

/// Halves an even term; otherwise triples it and adds one.
long long nextTerm(long long value) {
    return (value % 2 == 0) ? value / 2 : 3 * value + 1;
}

}  // namespace

int main() {
    long long start;
    std::cin >> start;

    // The sequence is deterministic, so the first repeated value is the point
    // at which it has entered a cycle.
    std::unordered_set<long long> seen;
    seen.insert(start);

    long long value = start;
    int index = 1;

    while (true) {
        value = nextTerm(value);
        ++index;
        if (!seen.insert(value).second) {
            std::cout << index << '\n';
            return 0;
        }
    }
}
